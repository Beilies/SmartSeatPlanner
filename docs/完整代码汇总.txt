# -*- coding: utf-8 -*-
"""
程序路径统一处理（app_paths.py）
==================================================================
为什么要单独写这个模块：
    老师用的免安装 exe 是用 PyInstaller 打包的，运行时会把 Python 和所有
    依赖解压到一个临时目录（_MEIxxxxx），程序退出后这个目录会被删掉。
    如果还用 __file__ 定位目录，「教室模板」「运行日志」「自检输出」就会
    写进临时目录然后丢失，老师下次打开会发现模板不见了。

    所以统一约定：
      · 源码运行时  BASE_DIR = 项目根目录（src/ 的上一级）
      · 免安装 exe 时 BASE_DIR = exe 文件所在目录（绿色软件习惯）
      · 如果 exe 放在只读位置（例如 C:\\Program Files），自动退回
        「我的文档\\SmartSeatPlanner」，保证程序永远有地方存数据。

    工程采用 src/ 布局：源码在 src/，图标在 assets/，模板与日志统一写 BASE_DIR，
    这样 `git clone` 下来的仓库根目录不会被运行产物弄脏（相关目录已在 .gitignore 里）。
"""
from __future__ import annotations

import os
import sys

# 是否被打包成了独立 exe（PyInstaller / cx_Freeze 都会设置 sys.frozen）
IS_FROZEN = bool(getattr(sys, "frozen", False))


def _project_root() -> str:
    """源码方式下定位工程根目录：src/ 的上一级，且应有 pyproject.toml。"""
    here = os.path.dirname(os.path.abspath(__file__))
    parent = os.path.dirname(here)
    if os.path.isfile(os.path.join(parent, "pyproject.toml")):
        return parent
    # 模块被单独拷走 / pip 安装到 site-packages 时，退回当前工作目录
    return os.getcwd()


def _can_write(path: str) -> bool:
    """用一个临时探测文件判断目录是否真的可写（比只看权限位可靠）。"""
    try:
        os.makedirs(path, exist_ok=True)
        probe = os.path.join(path, ".可写测试.tmp")
        with open(probe, "w", encoding="utf-8") as f:
            f.write("ok")
        os.remove(probe)
        return True
    except Exception:
        return False


def _resolve_base_dir() -> str:
    if IS_FROZEN:
        # sys.executable 在打包后就是 exe 本身的完整路径
        candidate = os.path.dirname(os.path.abspath(sys.executable))
    else:
        candidate = _project_root()
    if _can_write(candidate):
        return candidate

    # 只读目录的兜底：改存到用户文档下
    docs = os.path.join(os.path.expanduser("~"), "Documents", "SmartSeatPlanner")
    if _can_write(docs):
        return docs
    home = os.path.join(os.path.expanduser("~"), "SmartSeatPlanner")
    try:
        os.makedirs(home, exist_ok=True)
    except Exception:
        pass
    return home


BASE_DIR = _resolve_base_dir()

# 数据子目录（模板、自检输出都放在 BASE_DIR 下，跟着 exe 走）
TEMPLATE_DIRNAME = "教室模板"
OUTPUT_DIRNAME = "自检输出"
LOG_FILENAME = "运行日志.txt"


def asset_path(filename: str) -> str:
    """
    取程序内置资源（图标等）的路径。

    打包成单文件 exe 后，图标不在 exe 旁边，而是被 PyInstaller 解压到
    临时目录 sys._MEIPASS，所以不能用 BASE_DIR 去找；退出后临时目录会被
    删除，但程序运行期间读取是安全的。源码方式运行时资源在 assets/ 目录。
    找不到时返回第一个候选路径，由调用方判断文件是否存在并安静降级。
    """
    candidates = []
    if IS_FROZEN:
        meipass = getattr(sys, "_MEIPASS", "")
        if meipass:
            candidates.append(os.path.join(meipass, filename))
        candidates.append(os.path.join(
            os.path.dirname(os.path.abspath(sys.executable)), filename))
    else:
        root = _project_root()
        candidates.append(os.path.join(root, "assets", filename))
        # 兼容「图标直接放在源码目录 / 项目根目录」的旧布局
        candidates.append(os.path.join(root, "src", filename))
        candidates.append(os.path.join(root, filename))
    for path in candidates:
        if os.path.isfile(path):
            return path
    return candidates[0]

@dataclass
class Student:
    """一名学生。所有字段都允许为空，界面与算法必须容忍脏数据。"""
    sid: str = ""                                   # 学号
    name: str = ""                                  # 姓名
    score: Optional[float] = None                   # 总分；None = 缺失，绝不按 0 计
    gender: str = ""                                # 男 / 女 / 空（未知）
    tag: str = ""                                   # 特殊标记（自由文本）
    forbid: List[str] = field(default_factory=list)  # 禁止同桌 / 相邻的学生姓名
    subjects: Dict[str, Optional[float]] = field(default_factory=dict)  # 各科成绩
    source_row: int = 0                             # 原始 Excel 行号，便于老师核对

    # -- 判定 ----------------------------------------------------------
    def is_focus(self) -> bool:
        """是否为重点关注学生（浅红色）。只看特殊标记文本，规则简单可预期。"""
        t = (self.tag or "").strip()
        return any(k in t for k in FOCUS_KEYWORDS)

    def short_tag(self) -> str:
        """方格内显示的简短标记，太长会遮挡姓名。"""
        t = (self.tag or "").strip()
        if not t:
            return ""
        for k in FOCUS_KEYWORDS:
            if k in t:
                return k
        return t[:3]

    def forbid_text(self) -> str:
        return "、".join(self.forbid) if self.forbid else ""

    def score_text(self) -> str:
        return "缺考/未填" if self.score is None else f"{self.score:g}"

    def identity(self) -> str:
        """用于去重 / 黑名单匹配的唯一标识，优先学号，其次姓名。"""
        return (self.sid or self.name).strip()

    def clone(self) -> "Student":
        return Student(
            sid=self.sid, name=self.name, score=self.score, gender=self.gender,
            tag=self.tag, forbid=list(self.forbid), subjects=dict(self.subjects),
            source_row=self.source_row,
        )

    def __hash__(self) -> int:
        # 允许把 Student 放进 set / 作为字典键使用；内容相同的两个对象视为不同人时
        # 由 identity() 区分，这里只按身份哈希，避免可变字段影响哈希值。
        return hash((self.sid, self.name, self.source_row))

"""
Excel 导入模块（excel_io.py）
------------------------------------------------------------------
职责：
    1. 列出 Sheet；用 pandas + openpyxl(.xlsx) / xlrd(.xls) 读取成二维表
    2. 自动模糊匹配列名：学号 / 姓名 / 总分 / 性别 / 特殊标记 /
       禁止同桌名单 / 各科成绩
    3. 容错：表头不在第一行、合并单元格、空行、底部合计行、
       缺考与 N/A 等缺失值（绝不按 0 分）、异常分数、重复学生

    5. 预览窗口编辑后的表格文本 -> 学生对象（双向转换）

    成绩表经常前面有几行标题、说明、合并单元格，直接读会把说明当表头。

    再用 openpyxl / xlrd 把合并单元格的值补进矩阵，容错能力更强。
"""

def is_blank(value) -> bool:
    """判断是否为空值 / 缺考标记。注意：缺考标记不能按 0 分处理。"""
    if value is None:
        return True
    try:
        if pd.isna(value):
            return True
    except (TypeError, ValueError):
        pass
    return str(value).strip().lower() in MISSING_TOKENS


def as_text(value) -> str:
    """把单元格转成干净文本；NaN / None / 缺考标记统一返回空串。"""
    if is_blank(value):
        return ""
    if isinstance(value, float) and value.is_integer():
        return str(int(value))
    return str(value).strip()


def clean_score(value) -> Tuple[Optional[float], str]:
    """
    分数清洗：返回 (分数, 状态)，状态 = ok / missing / invalid
    - missing：空、—、N/A、缺考 等，返回 None（后续绝不当 0 分）
    - invalid：负数、超过 1000、无法识别的文本，返回 None 并告警
    """
    if is_blank(value):
        return None, "missing"
    if isinstance(value, bool):
        return None, "invalid"
    if isinstance(value, (int, float)):
        num = float(value)
    else:
        s = str(value).strip()
        # 全角数字 / 全角小数点 / 中文「分」/ 千分位逗号 的容错
        s = s.translate(str.maketrans("０１２３４５６７８９．－＋", "0123456789.-+"))
        s = s.replace("，", "").replace(",", "").replace("分", "").strip()
        if not re.fullmatch(r"[-+]?\d+(\.\d+)?", s):
            return None, "invalid"
        num = float(s)
    if num < 0 or num > 1000:
        return None, "invalid"
    return num, "ok"


def normalize_gender(value) -> str:
    s = as_text(value).lower()
    if s in ("男", "m", "male", "1", "男生", "男孩", "男同学"):
        return "男"
    if s in ("女", "f", "female", "2", "女生", "女孩", "女同学"):
        return "女"
    return ""


def split_forbid(value) -> List[str]:
    """禁止同桌名单：支持 顿号 / 逗号 / 分号 / 斜杠 / 空格 分隔。"""
    s = as_text(value)
    if not s or s.lower() in ("无", "没有", "none"):
        return []
    parts = re.split(r"[、,，;；/|\s]+", s)
    return [p for p in (x.strip() for x in parts) if p]


def parse_subjects_text(text: str) -> Dict[str, Optional[float]]:
    """预览表格里「语文=90;数学=85」这样的文本 -> 科目分数字典。"""
    out: Dict[str, Optional[float]] = {}
    if not text or not text.strip():
        return out
    for chunk in re.split(r"[;；,，\n]+", text):
        chunk = chunk.strip()
        if not chunk:
            continue
        m = re.match(r"^([^=:]+?)\s*[=:]\s*(.+)$", chunk)
        if not m:
            continue
        subject = m.group(1).strip()
        val, _ = clean_score(m.group(2).strip())
        if subject:
            out[subject] = val
    return out


def subjects_to_text(subjects: Dict[str, Optional[float]]) -> str:
    parts = []
    for k, v in subjects.items():
        parts.append(f"{k}={'缺考' if v is None else f'{v:g}'}")
    return ";".join(parts)

def is_supported_file(path: str) -> bool:
    return os.path.splitext(path)[1].lower() in (".xlsx", ".xls")


def list_sheets(path: str) -> List[str]:
    """列出所有 Sheet 名。文件损坏 / 缺依赖时抛中文错误。"""
    if not os.path.isfile(path):
        raise ExcelParseError("找不到这个 Excel 文件，请重新选择")
    if not is_supported_file(path):
        raise ExcelParseError("目前只支持 .xlsx 和 .xls 格式，请先另存为这两种格式")
    try:
        with pd.ExcelFile(path) as book:
            return list(book.sheet_names)
    except ImportError:
        raise ExcelParseError("读取 .xls 需要 xlrd 组件，请按使用说明安装依赖后重试")
    except Exception as exc:
        raise ExcelParseError(f"打开 Excel 失败：{_brief(exc)}")


def _brief(exc: Exception) -> str:
    """把底层异常压成一句中文可读的短说明，不暴露堆栈。"""
    msg = str(exc).strip().splitlines()[0] if str(exc).strip() else type(exc).__name__
    return msg[:110]


def read_matrix(path: str, sheet_name: str) -> pd.DataFrame:
    """读成「无表头」二维 DataFrame，并补齐合并单元格的值。"""
    ext = os.path.splitext(path)[1].lower()
    engine = "xlrd" if ext == ".xls" else "openpyxl"
    try:
        df = pd.read_excel(path, sheet_name=sheet_name, header=None,
                           engine=engine, dtype=object)
    except ImportError:
        raise ExcelParseError("缺少读取组件，请安装 xlrd（.xls）或 openpyxl（.xlsx）")
    except Exception as exc:
        raise ExcelParseError(f"读取工作表「{sheet_name}」失败：{_brief(exc)}")
    if df.empty:
        raise ExcelParseError(f"工作表「{sheet_name}」是空的，请检查文件内容")

    # --- 展开合并单元格：只有左上角格子有值，其余格子需补成同样的值 ---
    try:
        if ext == ".xlsx":
            wb = load_workbook(path, data_only=True, read_only=False)
            ws = wb[sheet_name] if sheet_name in wb.sheetnames else wb.worksheets[0]
            for rng in ws.merged_cells.ranges:
                value = ws.cell(rng.min_row, rng.min_col).value
                if value is None:
                    continue
                for r in range(rng.min_row, rng.max_row + 1):
                    for c in range(rng.min_col, rng.max_col + 1):
                        if r - 1 < df.shape[0] and c - 1 < df.shape[1]:
                            df.iat[r - 1, c - 1] = value
            wb.close()
        else:
            import xlrd  # .xls 专用
            book = xlrd.open_workbook(path, formatting_info=True)
            sh = book.sheet_by_name(sheet_name)
            for rlo, rhi, clo, chi in sh.merged_cells:
                value = sh.cell_value(rlo, clo)
                for r in range(rlo, rhi):
                    for c in range(clo, chi):
                        if r < df.shape[0] and c < df.shape[1]:
                            df.iat[r, c] = value
    except Exception:
        # 合并单元格信息读不到不影响主流程，静默降级（数据仍可用）
        pass
    return df


def detect_header(df: pd.DataFrame, max_scan: int = 20) -> Tuple[int, ColumnMapping]:
    """扫描前若干行，找出最像表头的一行，并做列名模糊匹配。"""
    best_row, best_score, best_mapping = -1, -1, None
    for r in range(min(max_scan, len(df))):
        texts = [as_text(df.iat[r, c]) for c in range(df.shape[1])]
        non_empty = [t for t in texts if t]
        if len(non_empty) < 2:
            continue
        mapping = _match_columns(texts)
        score = _header_score(texts, mapping)
        # 没有姓名列的行（例如大标题行）不可能是表头，直接降权
        if mapping.name is None:
            score -= 20
        if score > best_score:
            best_row, best_score, best_mapping = r, score, mapping

    if best_row < 0 or best_mapping is None:
        raise ExcelParseError("表格是空的或前 20 行里找不到表头")
    if best_mapping.name is None or best_score < 10:
        raise ExcelParseError(
            "没能识别出「姓名」列。请确认表头里有一列写着姓名（或学生姓名/Name），"
            "也可以先下载标准模板对照填写")
    return best_row, best_mapping


def _match_columns(texts: List[str]) -> ColumnMapping:
    """对一行文本做字段匹配，生成 ColumnMapping。"""
    mapping = ColumnMapping()
    occupied = set()
    for c, text in enumerate(texts):
        if not text:
            continue
        kind = None
        for key, pat in FIELD_PATTERNS:
            if pat.search(text):
                kind = key
                break
        if kind == "name" and "name" not in occupied:
            mapping.name = c
            occupied.add("name")
        elif kind == "sid" and mapping.sid is None:
            mapping.sid = c
        elif kind == "score" and mapping.score is None:
            mapping.score = c
        elif kind == "gender" and mapping.gender is None:
            mapping.gender = c
        elif kind == "tag" and mapping.tag is None:
            mapping.tag = c
        elif kind == "forbid" and mapping.forbid is None:
            mapping.forbid = c
        elif kind == "subject":
            mapping.subjects[c] = text
    # 未识别但全是数字、且列名不像「名次/班级」的列，也当作科目（兼容自定义科目名）
    return mapping


def _header_score(texts: List[str], mapping: ColumnMapping) -> int:
    weight = {"name": 10, "sid": 2, "score": 2, "gender": 1,
              "tag": 1, "forbid": 1, "subject": 1}
    score = 0
    for c, text in enumerate(texts):
        if not text:
            continue
        for key, pat in FIELD_PATTERNS:
            if pat.search(text):
                score += weight[key]
                break
    return score


def _guess_numeric_subjects(df: pd.DataFrame, header_row: int,
                            mapping: ColumnMapping) -> None:
    """把「表头没写科目名、但整列都是分数」的列补成科目列。"""
    n_rows = len(df)
    for c in range(df.shape[1]):
        if c in (mapping.sid, mapping.name, mapping.score, mapping.gender,
                 mapping.tag, mapping.forbid) or c in mapping.subjects:
            continue
        header_text = as_text(df.iat[header_row, c]) if header_row < n_rows else ""
        if NOT_SUBJECT.search(header_text):
            continue
        vals = [df.iat[r, c] for r in range(header_row + 1, n_rows)]
        vals = [v for v in vals if not is_blank(v)]
        if len(vals) < 2:
            continue
        numeric = sum(1 for v in vals if clean_score(v)[1] in ("ok", "missing"))
        if numeric / len(vals) >= 0.6:
            mapping.subjects[c] = header_text or f"科目{get_column_letter(c + 1)}"

# 二、矩阵 -> 学生对象
# ----------------------------------------------------------------------
def build_students(df: pd.DataFrame, header_row: int,
                   mapping: ColumnMapping) -> Tuple[List[Student], List[WarningItem], int, List[str]]:
    """把数据区逐行转成 Student，同时收集所有告警。"""
    if mapping.name is None:
        raise ExcelParseError("没有找到姓名列，无法生成学生名单")
    _guess_numeric_subjects(df, header_row, mapping)
    students: List[Student] = []
    warnings: List[WarningItem] = []
    ignored = 0
    auto_filled = 0                      # 总分由各科自动求和的人数
    subject_names = [mapping.subjects[c] for c in sorted(mapping.subjects)]

    def cell(r: int, c: Optional[int]):
        if c is None or c >= df.shape[1]:
            return ""
        return df.iat[r, c]

    for r in range(header_row + 1, len(df)):
        row_values = [df.iat[r, c] for c in range(df.shape[1])]
        if all(is_blank(v) for v in row_values):
            continue  # 空行

        line_no = r + 1  # Excel 里显示的行号
        sid = as_text(cell(r, mapping.sid))
        name = as_text(cell(r, mapping.name))

        # 底部合计 / 平均分 之类的统计行：直接忽略，不进入预览表。
        # 「合计」常写在学号列或姓名列，所以只检查学号/姓名/总分这三格，
        # 避免学生备注里出现「平均分」之类字样时被误删。
        stat_cells = (sid, name, as_text(cell(r, mapping.score)))
        if any(SUMMARY_ROW.search(x) for x in stat_cells) and \
                (not name or SUMMARY_ROW.search(name)):
            ignored += 1
            continue
        # 只有一个单元格有内容、又没有姓名/学号的杂项行，也忽略
        if not name and not sid and sum(1 for v in row_values if not is_blank(v)) <= 1:
            ignored += 1
            continue

        if not name:
            warnings.append(WarningItem("empty_name",
                                        "这一行姓名为空，已保留在预览表里，请补姓名或直接删除该行",
                                        line_no))

        # --- 各科成绩 ---
        subjects: Dict[str, Optional[float]] = {}
        absent_subjects: List[str] = []
        for c, sub_name in mapping.subjects.items():
            val, status = clean_score(cell(r, c))
            subjects[sub_name] = val
            if status == "missing":
                absent_subjects.append(sub_name)
            elif status == "invalid":
                warnings.append(WarningItem(
                    "bad_score", f"「{sub_name}」的值「{as_text(cell(r, c))}」不是有效分数，已按缺失处理（不按 0 分）",
                    line_no))

        # --- 总分：缺考/为空时，用有效科目分求和补齐，缺考科目不按 0 分 ---
        score, status = clean_score(cell(r, mapping.score)) if mapping.score is not None else (None, "missing")
        if status == "invalid":
            warnings.append(WarningItem(
                "bad_score", f"总分「{as_text(cell(r, mapping.score))}」不是有效分数，已按缺失处理", line_no))
        valid_scores = [v for v in subjects.values() if v is not None]
        if score is None and valid_scores:
            score = float(sum(valid_scores))
            auto_filled += 1
        if score is None:
            warnings.append(WarningItem(
                "absent", f"{name or '未填姓名'}：没有可用总分，排座时按班级平均水平处理（不按 0 分）", line_no))
        if absent_subjects:
            warnings.append(WarningItem(
                "absent", f"{name or '未填姓名'} 缺考/未填：{'、'.join(absent_subjects)}", line_no))

        students.append(Student(
            sid=sid, name=name, score=score,
            gender=normalize_gender(cell(r, mapping.gender)),
            tag=as_text(cell(r, mapping.tag)),
            forbid=split_forbid(cell(r, mapping.forbid)),
            subjects=subjects, source_row=line_no,
        ))

    # --- 整体说明类告警 ---
    if mapping.score is None:
        warnings.insert(0, WarningItem("info", "表格里没有「总分」列，已自动用各科成绩求和得到总分"))
    if auto_filled:
        warnings.insert(0, WarningItem(
            "info", f"共 {auto_filled} 人总分留空或写了缺考，已用有效科目成绩自动求和（缺考科目不按 0 分计）"))
    if subject_names:
        warnings.insert(0, WarningItem("info", f"识别到科目列：{'、'.join(subject_names)}"))
    warnings.insert(0, WarningItem(
        "info", f"表头位于第 {header_row + 1} 行，共读入 {len(students)} 名学生，忽略 {ignored} 行合计/无效行"))

    # --- 重复学生 ---
    for label, getter in (("学号", lambda s: s.sid), ("姓名", lambda s: s.name)):
        seen: Dict[str, int] = {}
        for s in students:
            key = getter(s).strip()
            if not key:
                continue
            seen[key] = seen.get(key, 0) + 1
        dup = [f"{k}({v}次)" for k, v in seen.items() if v > 1]
        if dup:
            warnings.append(WarningItem("duplicate",
                                        f"{label}重复：{'、'.join(dup[:8])}{'…' if len(dup) > 8 else ''}，"
                                        "请确认是否同一人，可在预览表里修改", 0))
    return students, warnings, ignored, subject_names

# -*- coding: utf-8 -*-
"""
教室布局模块（layout.py）
------------------------------------------------------------------
用二维字符网格描述教室：
    seat   = 学生座位（可以坐人）
    block  = 过道 / 垃圾桶 / 禁用空位（灰色，不能坐人）
    podium = 讲台所在格（深灰色，不能坐人；画布顶部另有整条讲台示意）

布局可以保存为 .json 模板，下次一键加载，老师不用重复点格子。

【扩展预留·仅注释，暂不实现】
    定时座位轮换：可在本模块增加 "rotation_group" 标记，
    把若干列划为同一轮换组，配合定时器整体平移 assignment 即可。
    小组功能：可给每个 seat 增加 group_id 字段，
    排座算法把「同组就近」加入成本函数即可，无需改动界面层。
"""
from __future__ import annotations

import json
import os
import re
from typing import Dict, Iterable, List, Tuple

# 单元格三种取值
SEAT = "seat"
BLOCK = "block"
PODIUM = "podium"
VALID_CELLS = (SEAT, BLOCK, PODIUM)

CELL_LABEL = {SEAT: "学生座位", BLOCK: "过道/禁用", PODIUM: "讲台"}
ZONE_NAMES = ("前排", "中排", "后排")

Pos = Tuple[int, int]  # (行, 列)，均从 0 开始


class LayoutError(Exception):
    """布局文件损坏 / 参数非法。上层 GUI 会捕获并转成中文弹窗。"""


class ClassroomLayout:
    """教室布局对象。所有坐标一律 (row, col)，从 0 开始。"""

    MIN_SIZE = 2
    MAX_SIZE = 30

    def __init__(self, rows: int = 6, cols: int = 9,
                 cells: List[List[str]] | None = None,
                 name: str = "默认教室"):
        self.name = name or "默认教室"
        self.rows = self._check_size(rows)
        self.cols = self._check_size(cols)
        if cells is None:
            self.cells = [[SEAT for _ in range(self.cols)] for _ in range(self.rows)]
        else:
            self.cells = self._normalize_cells(cells)
            self.rows, self.cols = len(self.cells), len(self.cells[0])

    # ------------------------------------------------------------------
    # 构造与校验
    # ------------------------------------------------------------------
    @staticmethod
    def _check_size(n: int) -> int:
        try:
            n = int(n)
        except (TypeError, ValueError):
            raise LayoutError("行列数必须是整数")
        if not (ClassroomLayout.MIN_SIZE <= n <= ClassroomLayout.MAX_SIZE):
            raise LayoutError(f"行列数请设置在 {ClassroomLayout.MIN_SIZE}~{ClassroomLayout.MAX_SIZE} 之间")
        return n

    def _normalize_cells(self, cells: List[List[str]]) -> List[List[str]]:
        """加载外部文件时做容错：非法字符一律降级为座位，行列不齐自动补齐。"""
        if not isinstance(cells, list) or not cells:
            raise LayoutError("布局文件内容为空")
        width = max((len(r) if isinstance(r, list) else 0) for r in cells)
        if width == 0:
            raise LayoutError("布局文件里没有有效的格子")
        width = min(max(width, self.MIN_SIZE), self.MAX_SIZE)
        out: List[List[str]] = []
        for row in cells[:self.MAX_SIZE]:
            safe = []
            for c in range(width):
                v = row[c] if isinstance(row, list) and c < len(row) else SEAT
                safe.append(v if v in VALID_CELLS else SEAT)
            out.append(safe)
        if not out:
            raise LayoutError("布局文件里没有有效的格子")
        return out

    @classmethod
    def default(cls, rows: int = 7, cols: int = 11) -> "ClassroomLayout":
        """
        默认教室布局（老师指定的常用教室）：
            7 行 × 11 列，3 个座位 | 过道 | 3 个座位 | 过道 | 3 个座位
            即第 4 列与第 8 列（从 1 数起）是纵向过道，共 7×9 = 63 个座位。
            画布顶部另有整条「讲台（前方）」示意，不占用网格格子。

        传入其它行列数时保持通用规则：宽教室 (>=6 列) 在中间留一条过道，
        这样旧模板 / 自检脚本指定 6×9 之类的尺寸时行为不变。
        """
        lay = cls(rows=rows, cols=cols, name="默认教室")
        if cols == 11:
            aisle_cols = (3, 7)          # 0 起算，对应界面上的第 4、8 列
        elif cols >= 6:
            aisle_cols = (cols // 2,)
        else:
            aisle_cols = ()
        for r in range(lay.rows):
            for c in aisle_cols:
                lay.cells[r][c] = BLOCK
        return lay

    def resize(self, rows: int, cols: int, preserve: bool = True) -> None:
        """改变行列数；preserve=True 时尽量保留左上角已有格子。"""
        rows, cols = self._check_size(rows), self._check_size(cols)
        old = [list(r) for r in self.cells]
        self.rows, self.cols = rows, cols
        self.cells = [[SEAT for _ in range(cols)] for _ in range(rows)]
        if preserve:
            for r in range(min(rows, len(old))):
                for c in range(min(cols, len(old[r]))):
                    self.cells[r][c] = old[r][c]

    def set_cell(self, row: int, col: int, value: str) -> None:
        if not (0 <= row < self.rows and 0 <= col < self.cols):
            return
        self.cells[row][col] = value if value in VALID_CELLS else SEAT

    def get(self, row: int, col: int) -> str:
        if 0 <= row < self.rows and 0 <= col < self.cols:
            return self.cells[row][col]
        return BLOCK

    def fill_all(self, value: str = SEAT) -> None:
        self.cells = [[value if value in VALID_CELLS else SEAT
                       for _ in range(self.cols)] for _ in range(self.rows)]

    # ------------------------------------------------------------------
    # 几何查询（供排座算法 / 导出 / 绘图共用）
    # ------------------------------------------------------------------
    def seats(self) -> List[Pos]:
        """按「从前往后、从左到右」返回所有可坐座位。"""
        return [(r, c) for r in range(self.rows) for c in range(self.cols)
                if self.cells[r][c] == SEAT]

    def desk_pairs(self) -> List[Tuple[Pos, Pos]]:
        """
        同桌对：同一行里连续两个 seat 组成一对，遇到过道/讲台就断开。
        例：座位 座位 过道 座位 座位 座位 座位 -> (0,1)、(3,4)、(5,6)
        这也是「强弱帮扶」「同水平就近」策略做初始配对时的依据。
        """
        pairs: List[Tuple[Pos, Pos]] = []
        for r in range(self.rows):
            run: List[int] = []
            for c in range(self.cols + 1):
                is_seat = c < self.cols and self.cells[r][c] == SEAT
                if is_seat:
                    run.append(c)
                else:
                    for i in range(0, len(run) - 1, 2):
                        pairs.append(((r, run[i]), (r, run[i + 1])))
                    run = []
        return pairs

    def deskmate(self, pos: Pos) -> Pos | None:
        """返回该座位的同桌座位（没有同桌返回 None）。"""
        for a, b in self.desk_pairs():
            if pos == a:
                return b
            if pos == b:
                return a
        return None

    def neighbors(self, pos: Pos) -> List[Pos]:
        """上下左右四邻中可坐人的座位；黑名单约束就检查这四个方向。"""
        r, c = pos
        out = []
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nr, nc = r + dr, c + dc
            if 0 <= nr < self.rows and 0 <= nc < self.cols and self.cells[nr][nc] == SEAT:
                out.append((nr, nc))
        return out

    def adjacent_pairs(self) -> List[Tuple[Pos, Pos]]:
        """所有共享边的座位对（每个对只出现一次）。"""
        pairs = []
        for r in range(self.rows):
            for c in range(self.cols):
                if self.cells[r][c] != SEAT:
                    continue
                if c + 1 < self.cols and self.cells[r][c + 1] == SEAT:
                    pairs.append(((r, c), (r, c + 1)))
                if r + 1 < self.rows and self.cells[r + 1][c] == SEAT:
                    pairs.append(((r, c), (r + 1, c)))
        return pairs

    def zone(self, row: int) -> str:
        """行号 -> 前排 / 中排 / 后排（按行数三等分）。"""
        if self.rows <= 0:
            return ZONE_NAMES[0]
        idx = min(2, (row * 3) // self.rows)
        return ZONE_NAMES[idx]

    def zone_rows(self) -> Dict[str, List[int]]:
        out = {z: [] for z in ZONE_NAMES}
        for r in range(self.rows):
            out[self.zone(r)].append(r)
        return out

    def front_ratio(self, row: int) -> float:
        """越靠前越接近 1，最后一排为 0。"""
        if self.rows <= 1:
            return 1.0
        return 1.0 - row / (self.rows - 1)

    def is_near_aisle(self, pos: Pos) -> bool:
        """是否靠过道：贴着教室左右边界，或左右相邻格是过道/讲台。"""
        r, c = pos
        if c == 0 or c == self.cols - 1:
            return True
        return (self.cells[r][c - 1] != SEAT) or (self.cells[r][c + 1] != SEAT)

    def summary(self) -> str:
        n_seat = len(self.seats())
        n_block = sum(v == BLOCK for row in self.cells for v in row)
        n_podium = sum(v == PODIUM for row in self.cells for v in row)
        return (f"{self.rows} 行 × {self.cols} 列｜可用座位 {n_seat} 个｜"
                f"过道/禁用 {n_block} 格｜讲台 {n_podium} 格｜同桌 {len(self.desk_pairs())} 对")

    # ------------------------------------------------------------------
    # JSON 模板存取
    # ------------------------------------------------------------------
    def to_dict(self) -> dict:
        return {"version": 1, "name": self.name, "rows": self.rows,
                "cols": self.cols, "cells": [list(r) for r in self.cells]}

    @classmethod
    def from_dict(cls, data: dict) -> "ClassroomLayout":
        if not isinstance(data, dict) or "cells" not in data:
            raise LayoutError("这不是有效的教室模板文件")
        rows = int(data.get("rows") or len(data["cells"]))
        cols = int(data.get("cols") or len(data["cells"][0]))
        return cls(rows=rows, cols=cols, cells=data["cells"],
                   name=str(data.get("name") or "导入的教室"))

    def save(self, path: str) -> None:
        folder = os.path.dirname(os.path.abspath(path))
        if folder:
            os.makedirs(folder, exist_ok=True)
        with open(path, "w", encoding="utf-8") as f:
            json.dump(self.to_dict(), f, ensure_ascii=False, indent=2)

    @classmethod
    def load(cls, path: str) -> "ClassroomLayout":
        try:
            with open(path, "r", encoding="utf-8") as f:
                data = json.load(f)
        except (OSError, json.JSONDecodeError) as exc:
            raise LayoutError(f"模板读取失败或格式不正确（{type(exc).__name__}）")
        return cls.from_dict(data)

    @staticmethod
    def template_dir(base_dir: str) -> str:
        """统一的模板目录：<项目目录>/教室模板"""
        return os.path.join(base_dir, "教室模板")

    @staticmethod
    def list_templates(base_dir: str) -> List[str]:
        """列出历史模板（返回完整路径，按文件名排序）。"""
        folder = ClassroomLayout.template_dir(base_dir)
        if not os.path.isdir(folder):
            return []
        names = [n for n in os.listdir(folder) if n.lower().endswith(".json")]
        return [os.path.join(folder, n) for n in sorted(names)]

    @staticmethod
    def safe_filename(name: str) -> str:
        """把老师输入的模板名转成安全文件名。"""
        cleaned = re.sub(r'[\\/:*?"<>|\r\n\t]', "_", (name or "").strip())
        return (cleaned or "我的教室")[:40]

# -*- coding: utf-8 -*-
"""
排座算法模块（seating.py）
==================================================================
【单独说明算法逻辑，方便后续改规则】

一、整体思路
    本模块不用「一次性贪心排完就结束」的做法，而是：
        初始解（体现方案差异） -> 成本函数打分 -> 随机交换 + 模拟退火局部优化
    这样既能做出 3~5 套风格不同的方案，又能保证每套方案都尽量满足硬约束。

二、成本函数 cost（越小越好）—— 想改规则，改 _cost() 里对应项即可
    1. 分区均衡     ：前排/中排/后排的「标准化平均分」差异越大，罚分越大。
                      标准化 = (个人总分 - 班级均分) / 标准差，避免不同科目满分不同带来的量纲问题。
    2. 优生打散     ：相邻两个座位都是优生（分数前 25%）时罚分，权重由左侧滑块控制。
                      滑块拉到最大 = 坚决不让优生扎堆；拉到最小 = 基本不管扎堆。
    3. 强弱搭配/同水平：
                      help 策略奖励同桌分差大（一强一弱，方便帮扶）；
                      similar 策略惩罚同桌分差大（水平接近，方便分层教学）；
                      其余策略不干预同桌分差。
    4. 黑名单       ：禁止同桌名单里的两人出现在上下左右相邻座位时重罚（权重远大于其它项）。
                      同时额外做一次「强制修复」，实在修不好就记录冲突并给老师降级建议，不崩溃。
    5. 性别均衡     ：开关打开时，同桌同性别要给一点罚分（只约束同桌，不强行拆散整排）。
    6. 特殊学生位置 ：重点关注学生坐得越靠前、越靠过道，成本越低（相当于奖励）。
                      front/aisle 两个指标都由布局模块计算，改布局即自动生效。
    7. 缺考/无总分  ：这类学生标准化分取 0（班级平均水平），绝不按 0 分拉低班级均分。

三、5 套方案怎么做到「差异化」
    方案1 强弱帮扶   ：高低分交替 + 同桌对优先，优化时奖励大分差同桌
    方案2 同水平就近 ：按分数排序后轮转分到三个区，区内水平接近，分区均分又均衡
    方案3 均衡随机   ：随机初始解 + 强分区均衡，座位关系最"松"
    方案4 性别优先   ：男女交替初始解 + 加大的性别权重（性别开关关闭时退化为均衡随机）
    方案5 特殊学生优先：重点关注学生先占「靠前 + 靠过道」最佳位置，其余随机 + 均衡
    每套方案用不同随机种子；万一生成结果与前面某套完全一样，会自动换种子重跑。

四、约束冲突时的降级策略（全部转成中文提示，绝不抛异常给老师）
    - 学生数 > 座位数   ：优先安排有特殊标记的学生，其余放进「未安排名单」，提示扩大行列数
    - 学生数 < 座位数   ：只启用「从前往后、从左到右」的前 N 个座位，
                          空位统一留在最后几排的末尾，绝不夹在学生中间
    - 可用座位为 0      ：抛 SeatingError，主界面弹窗建议打开布局编辑器
    - 黑名单无解        ：先尝试强制交换修复；仍冲突则保留方案 + 列出冲突名单 + 给出放宽建议
    - 名单里的名字找不到：不算错误，只在提示里说明「未匹配到该学生」
    - 班级人数太少/分数全相同：标准差按 1 处理，避免除零

五、扩展预留（仅注释，暂不实现）
    【定时座位轮换】可在 Plan 上增加 effective_date / next_rotate 字段，
      按 layout 的列分组做整体左右平移，复用 _optimize() 做微调即可。
    【小组功能】给座位增加 group_id，成本函数里加一项「同组学生尽量相邻」
      （w_group * 同组相邻对数取负），再把小组名显示到方格角标即可。
"""

# ----------------------------------------------------------------------
# 上下文：把 repeated 计算预先做好，优化循环里只做纯数值运算
# ----------------------------------------------------------------------
class _Context:
    def __init__(self, students: List[Student], layout: ClassroomLayout,
                 options: SeatingOptions):
        self.layout = layout
        self.options = options
        self.seats = layout.seats()
        self.desk_pairs = layout.desk_pairs()
        self.adjacent_pairs = layout.adjacent_pairs()

        # 人比座位少时，只在前 N 个座位里排，空位统一留在后排末尾。
        # 优化器只在「已入座」的座位之间交换，所以空位位置一旦定下就不会再散开。
        self.used_seats = _front_seats(self.seats, len(students))
        used_set = set(self.used_seats)
        self.used_desk_pairs = [pair for pair in self.desk_pairs
                                if pair[0] in used_set and pair[1] in used_set]

        scores = [s.score for s in students if s.score is not None]
        self.mean = statistics.fmean(scores) if scores else 0.0
        try:
            self.std = statistics.pstdev(scores) if len(scores) > 1 else 1.0
        except statistics.StatisticsError:
            self.std = 1.0
        if self.std < 1e-6:
            self.std = 1.0  # 全班同分，避免除零

        # 优生 / 学困生阈值：分数前 25% / 后 25%（无分数者不算）
        ordered = sorted(scores)
        if ordered:
            self.high_cut = ordered[int(0.75 * (len(ordered) - 1))]
            self.low_cut = ordered[int(0.25 * (len(ordered) - 1))]
        else:
            self.high_cut, self.low_cut = float("inf"), float("-inf")

        self.norm: Dict[str, float] = {}       # 姓名 -> 标准化分
        self.high: Dict[str, bool] = {}
        self.low: Dict[str, bool] = {}
        self.focus: Dict[str, bool] = {}
        self.gender: Dict[str, str] = {}
        for s in students:
            key = s.name
            self.norm[key] = 0.0 if s.score is None else (s.score - self.mean) / self.std
            self.high[key] = s.score is not None and s.score >= self.high_cut
            self.low[key] = s.score is not None and s.score <= self.low_cut
            self.focus[key] = s.is_focus()
            self.gender[key] = s.gender

        # 黑名单：把「姓名对」放进集合，双向都能查
        self.ban_pairs = set()
        by_name: Dict[str, Student] = {}
        for s in students:
            by_name.setdefault(s.name.strip(), s)
        self.unmatched_bans: List[str] = []
        for s in students:
            for target in s.forbid:
                t = by_name.get(target.strip())
                if t is None or t.name == s.name:
                    self.unmatched_bans.append(f"{s.name} → {target}")
                    continue
                self.ban_pairs.add(frozenset((s.name.strip(), t.name.strip())))

        # 特殊学生位置的「理想程度」预计算（越靠前、越靠过道越接近 1）
        self.position_value: Dict[Pos, float] = {}
        for pos in self.seats:
            front = layout.front_ratio(pos[0])
            aisle = 1.0 if layout.is_near_aisle(pos) else 0.0
            self.position_value[pos] = 0.6 * front + 0.4 * aisle

    def is_banned(self, a: Student, b: Student) -> bool:
        if not self.options.use_blacklist:
            return False
        return frozenset((a.name.strip(), b.name.strip())) in self.ban_pairs


# ----------------------------------------------------------------------
# 成本函数
# ----------------------------------------------------------------------
def _cost(assign: Assignment, ctx: _Context, cfg: dict) -> float:
    """计算一套座位方案的「不满意程度」，数值越小越好。"""
    cost = 0.0
    spread_w = 0.3 + 3.0 * ctx.options.spread_weight   # 优生打散权重（滑块）

    # ① 分区成绩均衡：三个区的标准化均分尽量都接近 0（班级平均）
    zone_sum = {z: 0.0 for z in ZONE_NAMES}
    zone_cnt = {z: 0 for z in ZONE_NAMES}
    for pos, s in assign.items():
        z = ctx.layout.zone(pos[0])
        zone_sum[z] += ctx.norm.get(s.name, 0.0)
        zone_cnt[z] += 1
    for z in ZONE_NAMES:
        avg = zone_sum[z] / zone_cnt[z] if zone_cnt[z] else 0.0
        cost += cfg["w_zone"] * abs(avg)

    # ② 优生打散：上下左右相邻的两个座位都是优生 -> 罚分
    for a, b in ctx.adjacent_pairs:
        sa, sb = assign.get(a), assign.get(b)
        if sa is None or sb is None:
            continue
        if ctx.high.get(sa.name) and ctx.high.get(sb.name):
            cost += spread_w * 1.2

    # ③ 同桌搭配偏好（help 奖励分差 / similar 惩罚分差）
    mode = cfg["pair_mode"]
    if mode != "none" and cfg["w_pair"] > 0:
        for a, b in ctx.desk_pairs:
            sa, sb = assign.get(a), assign.get(b)
            if sa is None or sb is None:
                continue
            diff = min(3.0, abs(ctx.norm.get(sa.name, 0.0) - ctx.norm.get(sb.name, 0.0)))
            if mode == "help":
                cost -= cfg["w_pair"] * diff * 0.35     # 分差越大越划算（帮扶）
            elif mode == "similar":
                cost += cfg["w_pair"] * diff * 0.35     # 分差越小越划算（同水平）
            # ④ 黑名单：同桌/相邻重罚（权重远高于其它软约束）
            if ctx.is_banned(sa, sb):
                cost += 40.0
            # ⑤ 性别均衡：同桌同性别轻微罚分
            if ctx.options.gender_balance and sa.gender and sa.gender == sb.gender:
                cost += 1.4 * cfg["w_gender"]
    else:
        # 即使不评价分差，黑名单与性别约束仍然要检查
        for a, b in ctx.desk_pairs:
            sa, sb = assign.get(a), assign.get(b)
            if sa is None or sb is None:
                continue
            if ctx.is_banned(sa, sb):
                cost += 40.0
            if ctx.options.gender_balance and sa.gender and sa.gender == sb.gender:
                cost += 1.4 * cfg["w_gender"]

    # 相邻但不同桌的黑名单同样禁止（前后排相邻也算「相邻」）
    if ctx.options.use_blacklist and ctx.ban_pairs:
        for a, b in ctx.adjacent_pairs:
            sa, sb = assign.get(a), assign.get(b)
            if sa is None or sb is None:
                continue
            if ctx.is_banned(sa, sb):
                cost += 12.0     # 同桌对已在上面重罚，这里避免重复计满权重

    # ⑥ 特殊学生优先靠前、靠过道：位置越理想，成本越低
    w_special = cfg["w_special"]
    if w_special:
        for pos, s in assign.items():
            if ctx.focus.get(s.name):
                cost -= w_special * ctx.position_value.get(pos, 0.0)

    return cost


def _swap(assign: Assignment, a: Pos, b: Pos) -> None:
    """交换两个座位上的学生（允许其中一个是空座位）。"""
    va, vb = assign.get(a), assign.get(b)
    if va is None and vb is None:
        return
    if va is None:
        assign[a] = vb
        assign.pop(b, None)
    elif vb is None:
        assign[b] = va
        assign.pop(a, None)
    else:
        assign[a], assign[b] = vb, va


def _banned_conflicts(assign: Assignment, ctx: _Context) -> List[Tuple[Pos, Pos]]:
    """列出仍然违反黑名单的相邻座位对。"""
    out = []
    for a, b in ctx.adjacent_pairs:
        sa, sb = assign.get(a), assign.get(b)
        if sa is not None and sb is not None and ctx.is_banned(sa, sb):
            out.append((a, b))
    return out


def _repair_blacklist(assign: Assignment, ctx: _Context, cfg: dict,
                      rng: random.Random, max_rounds: int = 12) -> int:
    """
    黑名单强制修复：对每一对冲突，尝试把其中一人换到别处。
    返回仍存在的冲突数量。修不好也不抛错，由上层转成中文降级提示。
    """
    occupied = [p for p in ctx.seats if p in assign]
    if len(occupied) < 2 or not ctx.ban_pairs:
        return len(_banned_conflicts(assign, ctx))
    for _ in range(max_rounds):
        conflicts = _banned_conflicts(assign, ctx)
        if not conflicts:
            return 0
        a, b = conflicts[0]
        base = _cost(assign, ctx, cfg)
        best, best_cost = None, base
        candidates = rng.sample(occupied, min(len(occupied), 30))
        for other in candidates:
            if other in (a, b):
                continue
            for source in (a, b):
                _swap(assign, source, other)
                c = _cost(assign, ctx, cfg)
                _swap(assign, source, other)   # 还原
                if c < best_cost - 1e-9:
                    best, best_cost = (source, other), c
        if best is None:
            break
        _swap(assign, best[0], best[1])
    return len(_banned_conflicts(assign, ctx))


# ----------------------------------------------------------------------
# 初始解：不同策略给出不同「起点」，这是方案差异化的主要来源
# ----------------------------------------------------------------------
def _front_seats(seats: List[Pos], count: int) -> List[Pos]:
    """只取「从前往后、从左到右」的前 count 个座位。

    【为什么不做均匀留白】早期版本会等间隔抽座位，导致学生中间夹着一个个
    「空位」，看起来像漏排了人，老师也不好打印发家长。现在人少于座位时，
    统一把多出来的座位留在最后几排末尾：学生连成一片，空位集中且不显眼。
    seats 必须已按 (行, 列) 排好序（ClassroomLayout.seats() 天然如此）。
    """
    if count >= len(seats):
        return list(seats)
    if count <= 0:
        return []
    return list(seats[:count])


def _initial_assignment(strategy: str, students: List[Student],
                        seats: List[Pos], ctx: _Context,
                        rng: random.Random) -> Assignment:
    """生成初始解。想调整某套方案的编排习惯，改这里即可。"""
    assign: Assignment = {}
    n = len(students)
    if n == 0 or not seats:
        return assign

    by_score = sorted(students, key=lambda s: (s.score is None, -(s.score or 0)))

    if strategy == "help":
        # 高低分交替：一强一弱尽量落在同一同桌对上
        half = n // 2
        high = by_score[:half]
        low = by_score[half:][::-1]      # 低分组倒序，让最强配最弱
        paired: List[Student] = []
        for i in range(max(len(high), len(low))):
            if i < len(high):
                paired.append(high[i])
            if i < len(low):
                paired.append(low[i])
        order_seats: List[Pos] = []
        for a, b in ctx.used_desk_pairs:
            order_seats.extend([a, b])
        for p in seats:
            if p not in order_seats:
                order_seats.append(p)
        for p, s in zip(order_seats, paired):
            assign[p] = s
        return assign

    if strategy == "similar":
        # 按分数排序后轮转分到三个区：区内水平接近，区间平均分又均衡
        zone_seats: Dict[str, List[Pos]] = {z: [] for z in ZONE_NAMES}
        for p in seats:
            zone_seats[ctx.layout.zone(p[0])].append(p)
        zone_students: Dict[str, List[Student]] = {z: [] for z in ZONE_NAMES}
        for i, s in enumerate(by_score):
            zone_students[ZONE_NAMES[i % 3]].append(s)
        for z in ZONE_NAMES:
            for p, s in zip(zone_seats[z], zone_students[z]):
                assign[p] = s
        # 若某个区座位不够（例如过道把前排切掉很多），把剩余学生放到空座位
        used_ids = {id(s) for s in assign.values()}
        left = [s for s in students if id(s) not in used_ids]
        free = [p for p in seats if p not in assign]
        for p, s in zip(free, left):
            assign[p] = s
        return assign

    if strategy == "gender":
        # 男女交替；未知性别的排在后面
        boys = [s for s in students if s.gender == "男"]
        girls = [s for s in students if s.gender == "女"]
        unknown = [s for s in students if s.gender not in ("男", "女")]
        rng.shuffle(boys); rng.shuffle(girls); rng.shuffle(unknown)
        merged: List[Student] = []
        for i in range(max(len(boys), len(girls))):
            if i < len(girls):
                merged.append(girls[i])
            if i < len(boys):
                merged.append(boys[i])
        merged.extend(unknown)
        for p, s in zip(_front_seats(seats, len(merged)), merged):
            assign[p] = s
        return assign

    if strategy == "front":
        # 重点关注学生先占「靠前 + 靠过道」的最佳座位
        focus = [s for s in students if s.is_focus()]
        others = [s for s in students if not s.is_focus()]
        rng.shuffle(others)
        best_seats = sorted(seats, key=lambda p: (-ctx.position_value.get(p, 0.0), p))
        used = set()
        for s, p in zip(focus, best_seats):
            assign[p] = s
            used.add(p)
        free = [p for p in _front_seats([p for p in seats if p not in used], len(others))]
        for p, s in zip(free, others):
            assign[p] = s
        return assign

    # random（默认）：完全随机起点，交给优化器做分区均衡与优生打散
    shuffled = list(students)
    rng.shuffle(shuffled)
    for p, s in zip(_front_seats(seats, len(shuffled)), shuffled):
        assign[p] = s
    return assign


def _optimize(assign: Assignment, ctx: _Context, cfg: dict,
              rng: random.Random, iterations: int) -> None:
    """
    模拟退火局部优化：随机挑两个有人坐的座位交换，成本下降就接受，
    偶尔也接受一点变差的解以跳出局部最优。温度逐步下降。
    """
    occupied = [p for p in ctx.seats if p in assign]
    if len(occupied) < 2:
        return
    cur = _cost(assign, ctx, cfg)
    temp = 0.9
    cooling = (0.02 / temp) ** (1.0 / max(1, iterations))
    for _ in range(iterations):
        a, b = rng.sample(occupied, 2)
        _swap(assign, a, b)
        new = _cost(assign, ctx, cfg)
        delta = new - cur
        if delta <= 0 or rng.random() < math.exp(-delta / max(temp, 1e-6)):
            cur = new
        else:
            _swap(assign, a, b)     # 不接受就还原
        temp *= cooling

def generate_plans(students: List[Student], layout: ClassroomLayout,
                   options: SeatingOptions) -> Tuple[List[Plan], List[str]]:
    """
    生成 3~5 套座位方案。
    返回 (方案列表, 全局提示)。全局提示里包含所有降级说明，GUI 会用中文弹窗展示。
    """
    notices: List[str] = []
    clean = [s for s in students if (s.name or "").strip()]
    if not clean:
        raise SeatingError("还没有有效学生，请先导入 Excel 成绩表")

    seats = layout.seats()
    if not seats:
        raise SeatingError("当前教室布局里一个学生座位都没有，请先打开「教室布局编辑器」标记座位")

    # —— 降级 1：人比座位多 ——
    placed = list(clean)
    unassigned: List[Student] = []
    if len(clean) > len(seats):
        # 有特殊标记的学生优先入座，其余保持原表顺序
        priority = [s for s in clean if s.is_focus()]
        rest = [s for s in clean if not s.is_focus()]
        placed = (priority + rest)[:len(seats)]
        placed_ids = {id(s) for s in placed}
        unassigned = [s for s in clean if id(s) not in placed_ids]
        notices.append(
            f"可用座位只有 {len(seats)} 个，但学生有 {len(clean)} 人。"
            f"已优先安排重点关注学生，还有 {len(unassigned)} 人未入座"
            f"（{ '、'.join(s.name for s in unassigned[:6]) }{'…' if len(unassigned) > 6 else ''}）。"
            "建议在布局编辑器里增加行列数，或减少过道格子。")
    elif len(clean) < len(seats):
        notices.append(f"学生 {len(clean)} 人、座位 {len(seats)} 个，"
                       f"多出的 {len(seats) - len(clean)} 个空位统一留在最后几排末尾，"
                       "不会夹在学生中间；老师也可以直接拖拽调整。")

    ctx = _Context(placed, layout, options)
    if ctx.unmatched_bans:
        sample = "、".join(ctx.unmatched_bans[:5])
        notices.append(f"禁止同桌名单里有 {len(ctx.unmatched_bans)} 条没匹配到学生（{sample}…），"
                       "已跳过这些记录；请核对姓名是否与名单完全一致。")
    if options.gender_balance and not any(s.gender for s in placed):
        notices.append("表格里没有可用的性别信息，「性别均衡」开关这次不会起作用（不影响其它规则）。")

    count = max(3, min(5, int(options.plan_count)))
    iterations = min(3000, 700 + 30 * len(placed))
    plans: List[Plan] = []
    signatures = set()

    for i in range(count):
        cfg = dict(STRATEGIES[i])
        key = cfg["key"]
        # 性别开关关闭时，方案4 自动退化成均衡随机版本，避免出现"名不副实"的方案
        if key == "gender" and not options.gender_balance:
            cfg["pair_mode"] = "none"
            cfg["w_gender"] = 0.0
            cfg["w_zone"] = 3.0
        best_assign = None
        for attempt in range(3):       # 结果撞车就换种子重跑
            rng = random.Random(options.seed + i * 7919 + attempt * 104729)
            assign = _initial_assignment(key, placed, ctx.used_seats, ctx, rng)
            _optimize(assign, ctx, cfg, rng, iterations)
            signature = tuple(sorted((s.name, p[0], p[1]) for p, s in assign.items()))
            if signature not in signatures:
                signatures.add(signature)
                best_assign = assign
                break
            if best_assign is None:
                best_assign = assign
        assert best_assign is not None

        # —— 降级 2：黑名单冲突尽力修复，修不好就如实告知 ——
        conflicts: List[str] = []
        remain = _repair_blacklist(best_assign, ctx, cfg, random.Random(options.seed + i + 1))
        if remain:
            pairs = _banned_conflicts(best_assign, ctx)[:4]
            detail = "；".join(
                f"{best_assign[a].name} 与 {best_assign[b].name}" for a, b in pairs
                if a in best_assign and b in best_assign)
            conflicts.append(f"仍有 {remain} 处黑名单相邻（{detail}）。"
                             "建议：减少禁止同桌名单人数、在两人之间加一格过道，"
                             "或生成更多方案后手动拖拽调整。")
            notices.append(f"{cfg['name']}：{conflicts[-1]}")

        plans.append(Plan(
            name=cfg["name"], strategy=key, description=cfg["desc"],
            assignment=best_assign, unassigned=list(unassigned),
            conflicts=conflicts,
        ))

    return plans, notices

# -*- coding: utf-8 -*-
"""
方案评估模块（evaluate.py）
------------------------------------------------------------------
把「这套座位好不好」变成老师一眼能看懂的两样东西：
    1. 前排 / 中排 / 后排 平均分（给主界面嵌入的柱状图用）
    2. 优生、学困生、重点关注学生分布 + 一句话中文结论（给文本面板用）

注意：缺考 / 无总分的学生不参与平均分计算（绝不按 0 分拉低均分），
      但在人数统计里仍然算「已入座」。
"""
from __future__ import annotations

from typing import Dict, List, Tuple

from layout import ClassroomLayout, ZONE_NAMES
from models import Student
from seating import Plan


def _thresholds(students: List[Student]) -> Tuple[float, float]:
    """优生 / 学困生分数阈值：有效成绩的前 25% / 后 25%。"""
    scores = sorted(s.score for s in students if s.score is not None)
    if not scores:
        return float("inf"), float("-inf")
    return (scores[int(0.75 * (len(scores) - 1))],
            scores[int(0.25 * (len(scores) - 1))])


def _ban_pairs(students: List[Student]) -> set:
    """重建禁止相邻名单（按姓名匹配，与排座算法口径一致）。"""
    names = {s.name.strip() for s in students}
    pairs = set()
    for s in students:
        for target in s.forbid:
            t = target.strip()
            if t in names and t != s.name.strip():
                pairs.add(frozenset((s.name.strip(), t)))
    return pairs


def evaluate_plan(plan: Plan, layout: ClassroomLayout,
                  students: List[Student]) -> dict:
    """计算单套方案的统计结果，返回结构化字典。"""
    assign = plan.assignment
    all_scores = [s.score for s in students if s.score is not None]
    overall = sum(all_scores) / len(all_scores) if all_scores else 0.0
    high_cut, low_cut = _thresholds(students)

    zone_scores: Dict[str, List[float]] = {z: [] for z in ZONE_NAMES}
    zone_counts: Dict[str, int] = {z: 0 for z in ZONE_NAMES}
    high_by_zone: Dict[str, int] = {z: 0 for z in ZONE_NAMES}
    low_by_zone: Dict[str, int] = {z: 0 for z in ZONE_NAMES}
    high_total = low_total = focus_total = focus_good = 0

    for pos, s in assign.items():
        z = layout.zone(pos[0])
        zone_counts[z] += 1
        if s.score is not None:
            zone_scores[z].append(s.score)
            if s.score >= high_cut:
                high_by_zone[z] += 1
                high_total += 1
            if s.score <= low_cut:
                low_by_zone[z] += 1
                low_total += 1
        if s.is_focus():
            focus_total += 1
            if z == "前排" or layout.is_near_aisle(pos):
                focus_good += 1

    zone_avgs = {z: (sum(v) / len(v) if v else overall) for z, v in zone_scores.items()}
    valid_avgs = list(zone_avgs.values())
    max_diff = (max(valid_avgs) - min(valid_avgs)) if valid_avgs else 0.0

    # 异性同桌比例
    hetero = same = 0
    for a, b in layout.desk_pairs():
        sa, sb = assign.get(a), assign.get(b)
        if sa is None or sb is None or not sa.gender or not sb.gender:
            continue
        if sa.gender != sb.gender:
            hetero += 1
        else:
            same += 1

    # 仍存在的黑名单相邻（拖拽换座后可能重新出现，所以每次都重算）
    bans = _ban_pairs(students)
    ban_hits = 0
    for a, b in layout.adjacent_pairs():
        sa, sb = assign.get(a), assign.get(b)
        if sa is not None and sb is not None and frozenset((sa.name.strip(), sb.name.strip())) in bans:
            ban_hits += 1

    # -- 一句话结论（给老师挑方案用） --
    if ban_hits:
        verdict = f"注意：还有 {ban_hits} 处黑名单相邻，建议手动拖拽或改用其它方案。"
    elif max_diff <= max(3.0, overall * 0.06):
        verdict = "各区成绩比较均衡，可以直接使用。"
    elif zone_avgs["前排"] > zone_avgs["后排"] + max(3.0, overall * 0.06):
        verdict = "前排平均分偏高，若想照顾后排可适当拖拽调整。"
    else:
        verdict = "后排平均分偏高，可看看重点关注学生是否都安排到了前排。"

    lines = [
        f"◆ {plan.name}｜{plan.description}",
        f"班级平均分 {overall:.1f} 分（有效成绩 {len(all_scores)} 人；"
        f"缺考/无总分 {len(students) - len(all_scores)} 人不计入平均分）",
        "前排 / 中排 / 后排 平均分：" +
        " / ".join(f"{zone_avgs[z]:.1f}" for z in ZONE_NAMES) +
        f"｜最大差 {max_diff:.1f} 分",
        f"优生 {high_total} 人：前排 {high_by_zone['前排']}、中排 {high_by_zone['中排']}、后排 {high_by_zone['后排']}",
        f"学困生 {low_total} 人：前排 {low_by_zone['前排']}、中排 {low_by_zone['中排']}、后排 {low_by_zone['后排']}",
        f"重点关注 {focus_total} 人，其中 {focus_good} 人坐在前排或靠过道",
        f"异性同桌 {hetero} 对 / 同性同桌 {same} 对｜黑名单相邻 {ban_hits} 处｜"
        f"未入座 {len(plan.unassigned)} 人",
        "结论：" + verdict,
    ]
    if plan.conflicts:
        lines.append("提示：" + plan.conflicts[0])

    return {
        "plan_name": plan.name,
        "overall_avg": overall,
        "zone_avgs": zone_avgs,
        "zone_counts": zone_counts,
        "max_diff": max_diff,
        "high_total": high_total, "low_total": low_total,
        "high_by_zone": high_by_zone, "low_by_zone": low_by_zone,
        "focus_total": focus_total, "focus_good": focus_good,
        "hetero": hetero, "same": same,
        "ban_hits": ban_hits,
        "unassigned": len(plan.unassigned),
        "verdict": verdict,
        "text": "\n".join(lines),
    }


def attach_stats(plans: List[Plan], layout: ClassroomLayout,
                 students: List[Student]) -> None:
    """生成方案或拖拽换座后调用：给每套方案刷新统计。"""
    for p in plans:
        p.stats = evaluate_plan(p, layout, students)


def compare_rows(plans: List[Plan]) -> Tuple[List[str], List[list]]:
    """所有方案的横向对比表（给「导出统计报表」用）。"""
    columns = ["方案", "策略说明", "前排均分", "中排均分", "后排均分",
               "区均分最大差", "优生人数", "学困生人数", "重点关注人数",
               "重点靠前/过道", "异性同桌对数", "黑名单相邻", "未入座人数", "结论"]
    rows = []
    for p in plans:
        st = p.stats or {}
        za = st.get("zone_avgs", {z: 0.0 for z in ZONE_NAMES})
        rows.append([
            p.name, p.description,
            round(za.get("前排", 0), 1), round(za.get("中排", 0), 1),
            round(za.get("后排", 0), 1), round(st.get("max_diff", 0), 1),
            st.get("high_total", 0), st.get("low_total", 0),
            st.get("focus_total", 0), st.get("focus_good", 0),
            st.get("hetero", 0), st.get("ban_hits", 0),
            st.get("unassigned", 0), st.get("verdict", ""),
        ])
    return columns, rows

# 刻意不画的「其它信息」：
#   · 方案标题、颜色图例、行列数、前排说明、导出时间
#   · 过道 / 讲台格：整列删掉，不画灰色方框、不写「过道」、也不留空白列
#   · 老师对学生的重点关照信息：不写「重点关注 / 视力 / 学困」等小字，
#     也不再用浅红底色区分——所有学生方格一律白色，家长看到的是同一张表
# 【注意】主界面画布仍保留浅红标记与灰色过道（老师排座时需要看），
#         只在这个「对外导出」的函数里做脱敏与简化；Excel 座位表也仍保留教师信息。
# ----------------------------------------------------------------------
def png_columns(layout: ClassroomLayout) -> List[int]:
    """返回 PNG 要画的原始列号：整列都是过道/讲台的列直接删掉。

    以「列」为单位压缩而不是逐格压缩，保证每一行仍然对齐；
    个别行里的过道格（其所在列其它行仍有座位）跳过不画即可。
    """
    return [c for c in range(layout.cols)
            if any(layout.get(r, c) == SEAT for r in range(layout.rows))]


def build_seat_figure(plan: Plan, layout: ClassroomLayout):
    """画出「纯座位信息」画布，返回 (fig, ax)。

    单独抽成函数是为了让自检脚本能直接检查图上到底画了哪些文字 / 颜色，
    防止以后改代码时又把标题、图例、过道方框或重点关注标记加回来。
    """
    rows = layout.rows
    draw_cols = png_columns(layout)
    out_cols = max(1, len(draw_cols))              # 极端容错：全无座位也不让坐标崩
    # 没有标题、页脚、过道列了，画布按实际网格收紧，避免大片空白
    fig_w = max(8.0, out_cols * 1.18 + 0.9)
    fig_h = max(4.0, rows * 0.78 + 1.15)
    fig, ax = plt.subplots(figsize=(fig_w, fig_h), dpi=180)
    fig.patch.set_facecolor("white")
    # 坐标紧贴内容：左边留一点给排号，上边留给讲台，下边不再留页脚空间
    ax.set_xlim(-0.60, out_cols + 0.10)
    ax.set_ylim(-0.06, rows + 1.00)
    ax.set_aspect("equal")
    ax.axis("off")
    fig.subplots_adjust(left=0.01, right=0.99, bottom=0.01, top=0.99)

    # 讲台横条（属于座位方位信息，保留），居中于压缩后的座位网格
    ax.add_patch(Rectangle((out_cols / 2 - 1.35, rows + 0.32), 2.7, 0.6,
                          facecolor=COLOR["podium"], edgecolor=COLOR["podium"]))
    ax.text(out_cols / 2, rows + 0.62, "讲　台", ha="center", va="center",
            fontsize=13, color="white", fontweight="bold")

    name_size = 11 if out_cols <= 10 else 9
    for r in range(rows):
        y = rows - 1 - r                           # 第 0 排画在最上方
        # 排号：帮助老师/家长定位第几排，属于座位信息
        ax.text(-0.18, y + 0.3, f"{r + 1}排", ha="right", va="center",
                fontsize=9, color=COLOR["subtext"])
        for x, c in enumerate(draw_cols):
            if layout.get(r, c) != SEAT:
                continue                           # 过道 / 讲台格：不画框、不写字
            stu = plan.assignment.get((r, c))
            if stu is None:
                fill, border = COLOR["empty"], COLOR["empty_border"]
                label, size, color = "空位", name_size - 1, "#9AA6B2"
            else:
                # 统一白色：不泄露「重点关注」等老师内部标记
                fill, border = COLOR["normal"], COLOR["normal_border"]
                label, size, color = stu.name, name_size, COLOR["text"]
            ax.add_patch(Rectangle((x + 0.04, y + 0.04), 0.92, 0.56,
                                  facecolor=fill, edgecolor=border, linewidth=1.1))
            ax.text(x + 0.5, y + 0.32, label, ha="center", va="center",
                    fontsize=size, color=color,
                    fontweight="bold" if stu is not None else "normal")
    return fig, ax


def export_canvas_png(path: str, plan: Plan, layout: ClassroomLayout) -> str:
    """把 build_seat_figure 画好的座位图存成 PNG，返回实际路径。"""
    try:
        path = _ensure_ext(path, ".png")
        fig, _ = build_seat_figure(plan, layout)
        folder = os.path.dirname(os.path.abspath(path))
        if folder:
            os.makedirs(folder, exist_ok=True)
        # tight + 很小的留白：图片边缘干净，打印不会被裁到字
        fig.savefig(path, dpi=180, bbox_inches="tight", pad_inches=0.06,
                    facecolor="white")
        plt.close(fig)
        return path
    except Exception as exc:
        raise ExportError(f"导出 PNG 图片失败：{str(exc)[:100]}")

def export_report_xlsx(path: str, plans: List[Plan], layout: ClassroomLayout,
                       students: List[Student], notices: Optional[List[str]] = None) -> str:
    try:
        path = _ensure_ext(path, ".xlsx")
        wb = Workbook()
        ws = wb.active
        ws.title = "方案对比"
        columns, rows = compare_rows(plans)
        for c, h in enumerate(columns, start=1):
            cell = ws.cell(row=1, column=c, value=h)
            cell.fill, cell.font, cell.alignment, cell.border = HEAD_FILL, HEAD_FONT, CENTER, GRID_BORDER

            for c, v in enumerate(row, start=1):

                cell.font, cell.alignment, cell.border = CELL_FONT, CENTER, GRID_BORDER
                if r % 2 == 0:
                    cell.fill = PatternFill("solid", fgColor="F4F7FA")
        ws.row_dimensions[1].height = 30
        ws.freeze_panes = "C2"
        widths = [20, 40, 12, 12, 12, 14, 11, 12, 14, 16, 14, 13, 12, 46]
        for c, w in enumerate(widths, start=1):
            ws.column_dimensions[get_column_letter(c)].width = w
        # 「区均分最大差」加数据条，越小越好，一眼看出哪套更均衡
        if rows:
            ws.conditional_formatting.add(
                f"F2:F{len(rows) + 1}",
                DataBarRule(start_type="num", start_value=0, end_type="max",
                            color="7FB3E8", showValue=True))

        # 分区分布明细
        detail = wb.create_sheet("分区分布")
        detail_cols = ["方案", "区域", "入座人数", "平均分", "优生人数", "学困生人数"]
        for c, h in enumerate(detail_cols, start=1):
            cell = detail.cell(row=1, column=c, value=h)
            cell.fill, cell.font, cell.alignment = HEAD_FILL, HEAD_FONT, CENTER
        rr = 2
        for p in plans:
            st = p.stats or {}
            for z in ZONE_NAMES:
                vals = [p.name, z, st.get("zone_counts", {}).get(z, 0),
                        round(st.get("zone_avgs", {}).get(z, 0), 1),
                        st.get("high_by_zone", {}).get(z, 0),
                        st.get("low_by_zone", {}).get(z, 0)]
                for c, v in enumerate(vals, start=1):
                    cell = detail.cell(row=rr, column=c, value=v)
                    cell.font, cell.alignment, cell.border = CELL_FONT, CENTER, GRID_BORDER
                rr += 1
        for c, w in enumerate([20, 10, 12, 12, 12, 14], start=1):
            detail.column_dimensions[get_column_letter(c)].width = w

        # 提示 / 冲突 / 降级记录
        note = wb.create_sheet("提示与冲突")
        note.cell(row=1, column=1, value="类型").fill = HEAD_FILL
        note.cell(row=1, column=2, value="内容").fill = HEAD_FILL
        for c in (1, 2):
            note.cell(row=1, column=c).font = HEAD_FONT
            note.cell(row=1, column=c).alignment = CENTER
        rr = 2
        lines = [(f"生成时间 {_stamp()}", "信息")]
        lines += [(n, "降级提示") for n in (notices or [])]
        for p in plans:
            lines += [(f"{p.name}：{c}", "约束冲突") for c in p.conflicts]
            if p.unassigned:
                lines.append((f"{p.name} 未入座：" + "、".join(s.name for s in p.unassigned),
                              "未入座名单"))
        for text, kind in lines:
            note.cell(row=rr, column=1, value=kind).font = CELL_FONT
            cell = note.cell(row=rr, column=2, value=text)
            cell.font, cell.alignment = CELL_FONT, Alignment(vertical="center", wrap_text=True)
            rr += 1
        note.column_dimensions["A"].width = 16
        note.column_dimensions["B"].width = 100

        folder = os.path.dirname(os.path.abspath(path))
        if folder:
            os.makedirs(folder, exist_ok=True)
        wb.save(path)
        return path
    except PermissionError:
        raise ExportError("报表文件正被 Excel 打开，请先关闭或换个文件名")
    except Exception as exc:
        raise ExportError(f"导出统计报表失败：{str(exc)[:100]}")

    def _hover(self, event) -> None:
        if self.drag and self.drag.get("active"):
            self._hide_tip()
            return
        pos = self._pos_at(event.x, event.y)
        if pos == self.hover_pos:
            if pos and self.tip.winfo_viewable():
                self.tip.geometry(f"+{event.x_root + 18}+{event.y_root + 14}")
            return
        self.hover_pos = pos
        if self.tip_job:
            try:
                self.after_cancel(self.tip_job)
            except Exception:
                pass
            self.tip_job = None
        if pos is None:
            self._hide_tip()
            return
        self.tip_job = self.after(280, lambda: self._show_tip(pos, event.x_root + 18, event.y_root + 14))

    def _show_tip(self, pos: Pos, rx: int, ry: int) -> None:
        if self.hover_pos != pos or self.plan is None:
            return
        self.tip_label.configure(text=self._detail_text(pos))
        self.tip.geometry(f"+{rx}+{ry}")
        self.tip.deiconify()
        self.tip.lift()

    def _hide_tip(self) -> None:
        self.hover_pos = None
        if self.tip_job:
            try:
                self.after_cancel(self.tip_job)
            except Exception:
                pass
            self.tip_job = None
        try:
            self.tip.withdraw()
        except Exception:
            pass

    # ------------------------------------------------------------------

    def _press(self, event) -> None:
        self._hide_tip()
        if self.plan is None:
            return
        pos = self._pos_at(event.x, event.y)
        if pos is None or self.layout.get(*pos) != SEAT:
            return
        self.drag = {"start": pos, "x": event.x, "y": event.y, "active": False}

    def _motion(self, event) -> None:
        if not self.drag:
            return
        if not self.drag["active"] and \
                abs(event.x - self.drag["x"]) + abs(event.y - self.drag["y"]) > 8:
            self.drag["active"] = True
            self._hide_tip()
        if not self.drag["active"]:
            return
        self.canvas.delete("ghost")
        w, h = self._cell_size()
        src = self.drag["start"]
        stu = self.plan.assignment.get(src)
        x, y = self.canvas.canvasx(event.x), self.canvas.canvasy(event.y)
        self.canvas.create_rectangle(x - w * 0.38, y - h * 0.34,
                                     x + w * 0.38, y + h * 0.34,
                                     fill=COLOR["ghost"], outline="#C79A20",
                                     width=2, tags="ghost")
        self.canvas.create_text(x, y, text=stu.name if stu else "空位",
                                font=("微软雅黑", max(8, int(10 * self.scale)), "bold"),
                                fill=COLOR["text"], tags="ghost")
        # 高亮即将放下的目标格
        target = self._pos_at(event.x, event.y)
        self.canvas.itemconfigure("seatrect", width=1.4)
        if target and self.layout.get(*target) == SEAT and target != src:
            self.canvas.itemconfigure(f"rect_{target[0]}_{target[1]}",
                                      outline=COLOR["select"], width=3)

    def _release(self, event) -> None:
        if not self.drag:
            return
        drag, self.drag = self.drag, None
        self.canvas.delete("ghost")
        self.canvas.itemconfigure("seatrect", width=1.4)
        if not drag["active"] or self.plan is None:
            return
        src = drag["start"]
        target = self._pos_at(event.x, event.y)
        if target is None or target == src or self.layout.get(*target) != SEAT:
            self.on_status("没有换座：请把学生拖到另一个学生座位或空位上")
            return
        self._push_undo(src, target)
        apply_swap(self.plan, src, target)
        self.draw()
        a = self.plan.assignment.get(src)
        b = self.plan.assignment.get(target)
        self.on_status(f"已换座：{src[0] + 1}排{src[1] + 1}列 ↔ "
                       f"{target[0] + 1}排{target[1] + 1}列"
                       f"（{(a.name if a else '空位')} / {b.name if b else '空位'}）｜Ctrl+Z 可撤销")
        if self.on_change:
            self.on_change()

    def _plan_key(self) -> str:
        return self.plan.name if self.plan else ""

    def _push_undo(self, a: Pos, b: Pos) -> None:
        stack = self.undo_stacks.setdefault(self._plan_key(), [])
        stack.append((a, b))
        if len(stack) > 60:
            stack.pop(0)

    def undo(self) -> None:
        """撤销上一步拖拽换座（每套方案各自记录）。"""
        if self.plan is None:
            return
        stack = self.undo_stacks.get(self._plan_key(), [])
        if not stack:
            self.on_status("这一步没有可撤销的换座操作")
            return
        a, b = stack.pop()
        apply_swap(self.plan, a, b)
        self.draw()
        self.on_status("已撤销上一步换座")
        if self.on_change:
            self.on_change()

    def _right_menu(self, event) -> None:
        self.right_pos = self._pos_at(event.x, event.y)
        if self.right_pos and self.layout.get(*self.right_pos) == SEAT:
            try:
                self.menu.tk_popup(event.x_root, event.y_root)
            finally:
                self.menu.grab_release()

    def _show_seat_info(self) -> None:
        """右键菜单：把悬浮提示的内容用弹窗再显示一次，方便投影讲解。"""
        if self.right_pos is None or self.plan is None:
            return
        try:
            messagebox.showinfo("座位信息", self._detail_text(self.right_pos), parent=self)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 缩放
    # ------------------------------------------------------------------
    def _set_scale(self, value: float) -> None:
        self.scale = min(1.8, max(0.55, value))
        self.draw()

    def zoom_in(self) -> None:
        self._set_scale(self.scale + 0.12)

    def zoom_out(self) -> None:
        self._set_scale(self.scale - 0.12)

    def fit(self) -> None:
        self.update_idletasks()
        cw = max(300, self.canvas.winfo_width())
        ch = max(240, self.canvas.winfo_height())
        sx = (cw - self.pad * 3) / ((self.layout.cols + 1.6) * self.BASE_W)
        sy = (ch - self.pad * 3) / ((self.layout.rows + 1.7) * self.BASE_H)
        self._set_scale(min(sx, sy))

    def _wheel(self, event) -> None:
        if event.state & 0x0004:      # Ctrl + 滚轮 -> 缩放
            self._set_scale(self.scale + (0.08 if event.delta > 0 else -0.08))
            return "break"
        self.canvas.yview_scroll(-1 if event.delta > 0 else 1, "units")

    def _setup_icon(self) -> None:
        """
        设置程序图标（老师自己设计的座位图）。

        三处都用同一份设计稿：
          1) exe 文件图标      —— 打包时由 --icon assets/app.ico 写入，资源管理器可见
          2) 标题栏 / 任务栏  —— iconbitmap 用 ico；iconphoto 用 png 做高 DPI 兜底
          3) 左侧面板左上角    —— 同一个 PhotoImage 直接放进标题行

        图标属于「锦上添花」：文件缺失或 Tk 不支持 PNG 时静默跳过，
        绝不让程序因为一个图片打不开，全部用 try 包住。
        """
        self._app_icon = None       # 必须长期持有引用，否则 Tk 会把图片回收掉
        try:
            ico = asset_path("app.ico")
            if os.path.isfile(ico):
                self.iconbitmap(ico)                 # Windows 窗口左上角小图标
        except Exception:
            pass
        try:
            png = asset_path("app_logo.png")
            if os.path.isfile(png):
                self._app_icon = tk.PhotoImage(file=png)
                # 第二个参数 default=True：预览窗、布局编辑器等子窗口也继承
                self.iconphoto(True, self._app_icon)
        except Exception:
            self._app_icon = None

    def _bind_panel_wheel(self) -> None:
        """
        让鼠标滚轮在左侧面板的「任意位置」都能上下滚动。

        原来只写了 canvas.bind("<MouseWheel>", ...)，这就是滚轮失灵的根因：
        Tk 把滚轮事件交给鼠标正下方的那个控件，而面板里全是按钮 / 标签 /
        输入框，它们的绑定链条是 [控件 → 控件类 → 顶层窗口 → all]，
        canvas 并不在链条里。于是指针停在按钮上、或停在右侧滚动条上时，
        滚轮事件根本到不了 canvas，看起来就像「滚动条拖得动、滚轮没反应」。

        两层保险：
          ① 递归给面板里每个控件（含滚动条）绑控件级事件，先于类绑定触发，
             顺带挡住 Spinbox / Combobox 被滚轮误改数值；
          ② 在主窗口（顶层绑定级别）再兜一次底，且先判断鼠标位置确实在
             左侧面板范围内、上方没有弹窗，避免影响右侧座位画布和预览表格。
        Windows 用 <MouseWheel>，Linux / macOS 触控板另兼容 <Button-4/5>。
        """
        outer, canvas = self.left_outer, self.left_canvas
        WHEEL_UNITS = 3            # 20 像素 × 3 = 一格滚轮约 60 像素

        def do_scroll(event) -> Optional[str]:
            # 弹窗（导入预览 / 布局编辑器）里的滚轮交给它们自己处理。
            # 注意：MouseWheel 事件的 num 不是数字，而是字符串 "??"，
            # 直接 int() 会抛异常让 Tk 静默丢掉整个绑定，必须先判类型。
            try:
                if event.widget.winfo_toplevel() is not self:
                    return None
                raw_num = getattr(event, "num", 0)
                num = raw_num if isinstance(raw_num, int) else 0
                try:
                    delta = int(getattr(event, "delta", 0) or 0)
                except (TypeError, ValueError):
                    delta = 0
                if num == 4:                      # Linux 向上滚
                    direction, notches = -1, 1
                elif num == 5:                    # Linux 向下滚
                    direction, notches = 1, 1
                elif delta:
                    direction = -1 if delta > 0 else 1
                    # Windows 一格 = 120；触控板的细碎滚动按一格算
                    notches = min(max(abs(delta) // 120, 1), 5)
                else:
                    return None
                canvas.yview_scroll(direction * WHEEL_UNITS * notches, "units")
                return "break"                    # 不让滚轮改到方案数量等输入值
            except (tk.TclError, AttributeError, TypeError, ValueError):
                return None

        def global_scroll(event) -> Optional[str]:
            """顶层兜底：只接管鼠标确实停在左侧控制面板上的滚轮事件。"""
            try:
                if event.widget.winfo_toplevel() is not self:
                    return None
                under = self.winfo_containing(event.x_root, event.y_root)
                if under is None or under.winfo_toplevel() is not self:
                    return None
                x0, y0 = outer.winfo_rootx(), outer.winfo_rooty()
                if not (x0 <= event.x_root < x0 + outer.winfo_width()
                        and y0 <= event.y_root < y0 + outer.winfo_height()):
                    return None
            except (tk.TclError, AttributeError, TypeError, ValueError):
                return None
            return do_scroll(event)

        def bind_tree(widget) -> None:
            for seq in ("<MouseWheel>", "<Button-4>", "<Button-5>"):
                try:
                    widget.bind(seq, do_scroll, add="+")
                except tk.TclError:
                    pass                # 个别平台不支持某个事件序列时跳过
            for child in widget.winfo_children():
                bind_tree(child)

        bind_tree(outer)
        for seq in ("<MouseWheel>", "<Button-4>", "<Button-5>"):
            self.bind(seq, global_scroll, add="+")

    def _tk_exception(self, exc, value, tb) -> None:
        """Tk 回调里的漏网异常也统一转成中文提示 + 日志。"""
        err = value if isinstance(value, BaseException) else Exception(str(value))
        detail = "".join(traceback.format_exception(exc, value, tb))
        log_exception(err, where="Tk 回调", detail=detail)
        show_error(self, err, where="Tk 回调")

    @safe_call
    def _import_students(self) -> None:
        path = filedialog.askopenfilename(
            parent=self, title="选择学生成绩 Excel",
            filetypes=[("Excel 文件", "*.xlsx *.xls"), ("所有文件", "*.*")])
        if not path:
            return
        sheets = list_sheets(path)
        sheet = sheets[0]
        if len(sheets) > 1:
            dlg = SheetChoiceDialog(self, sheets)
            if not dlg.result:
                return
            sheet = dlg.result
        result = import_excel(path, sheet)
        ImportPreviewDialog(self, result, self._on_import_confirm)
        self.set_status(f"已选择文件：{os.path.basename(path)}（{sheet}），请在预览窗口核对")

                           warnings: List[WarningItem]) -> None:

        self.import_warnings = warnings

        self.notices = []

                              f"重点关注 {sum(1 for s in students if s.is_focus())} 人｜"
                              f"无总分 {sum(1 for s in students if s.score is None)} 人")

        self.eval_panel.show_empty("学生名单已确认。点左侧「生成座位方案（F5）」，"

        self.set_status(f"导入完成，共 {len(students)} 人，请点「生成座位方案」")
        if warnings:
            show_info_list(self, "导入提醒",
                           [w.display() for w in warnings[:30]],
                           subtitle="以下记录供老师核对（预览表里已经可以修改）：")

def fit_dialog(win, pref_w: int, pref_h: int,
               min_w: int = 520, min_h: int = 360) -> None:
    """
    让弹窗适应老师电脑的屏幕并居中，避免按钮跑到屏幕外。

    为什么要单独处理：老式 1366×768 笔记本、或系统显示缩放 125%/150% 时，
    写死 1080×720 的窗口会高过屏幕可用区域，窗口最下面的「确认导入」按钮
    就看不到了，老师会卡在预览这一步。这里按屏幕尺寸留出任务栏/标题栏的
    余量再决定窗口大小，同时把 minsize 也限制在屏幕内。

    注意：Tk 的 pack 按「添加顺序」分配空间，最后才添加的底部按钮在空间
    不足时会被裁掉。所以调用方必须先 pack(side=tk.BOTTOM) 按钮区，
    再 pack 可伸缩的表格/画布，让空间不够时优先压缩内容区而不是按钮。
    """
    try:
        win.update_idletasks()
        sw, sh = win.winfo_screenwidth(), win.winfo_screenheight()
        avail_w = max(560, sw - 80)       # 左右留边
        avail_h = max(340, sh - 100)      # 上下留任务栏 / 标题栏 / 边框
        w = min(pref_w, avail_w)
        h = min(pref_h, avail_h)
        win.minsize(min(min_w, avail_w), min(min_h, avail_h))
        x = max(0, (sw - w) // 2)
        y = max(0, (sh - h) // 2 - 24)
        win.geometry(f"{w}x{h}+{x}+{y}")
    except Exception:
        try:
            win.geometry(f"{pref_w}x{pref_h}")
            win.minsize(min_w, min_h)
        except Exception:
            pass


def log_exception(exc: Exception, where: str = "", detail: str = "") -> None:
    """把完整堆栈写入日志文件；界面只显示友好中文。"""
    try:
        text = detail or traceback.format_exc()
        if not text or text.startswith("NoneType: None"):
            text = f"{type(exc).__name__}: {exc}\n"
        with open(LOG_FILE, "a", encoding="utf-8") as f:
            f.write(f"\n===== {datetime.now():%Y-%m-%d %H:%M:%S} {where} =====\n")
            f.write(text)
    except Exception:
        pass


def show_error(parent, exc: Exception, where: str = "") -> None:
    """统一的友好报错弹窗：自定义业务异常显示原因，其它异常只给通用建议。"""
    log_exception(exc, where)
    known = ("ExcelParseError", "SeatingError", "LayoutError", "ExportError")
    if type(exc).__name__ in known and str(exc).strip():
        text = str(exc)
    else:
        text = ("刚才这一步没能完成，常见原因：文件正被 Excel 打开、文件损坏、"
                "数据为空或模板格式不对。\n\n可以换一个文件重试；详细记录已写入「运行日志.txt」，"
                "发给技术人员即可定位问题。")
    try:
        messagebox.showerror("操作未完成", text, parent=parent)
    except Exception:
        pass


def safe_call(func):
    """给所有按钮回调套一层保护：任何异常都变成中文弹窗，绝不崩溃退出。"""
    @functools.wraps(func)
    def wrapper(self, *args, **kwargs):
        try:
            return func(self, *args, **kwargs)
        except Exception as exc:
            try:
                parent = self.winfo_toplevel()
            except Exception:
                parent = None
            show_error(parent, exc, where=getattr(func, "__qualname__", "界面操作"))
            return None
    return wrapper

class EditableTable(ttk.Frame):
    """带滚动条、可双击单元格修改、可删除/新增行的 Treeview 封装。"""

    def __init__(self, master, columns: List[str], rows: List[List[str]],
                 widths: Optional[List[int]] = None, height: int = 14):
        super().__init__(master)
        self.columns = list(columns)
        self.tree = ttk.Treeview(self, columns=self.columns, show="headings",
                                 height=height, selectmode="extended")
        default_widths = [110, 100, 90, 70, 130, 180, 230]
        for i, col in enumerate(self.columns):
            self.tree.heading(col, text=col)
            self.tree.column(col, width=(widths or default_widths)[i] if i < len(widths or default_widths) else 120,
                             anchor=tk.CENTER, stretch=True)
        vsb = ttk.Scrollbar(self, orient=tk.VERTICAL, command=self.tree.yview)
        hsb = ttk.Scrollbar(self, orient=tk.HORIZONTAL, command=self.tree.xview)
        self.tree.configure(yscrollcommand=vsb.set, xscrollcommand=hsb.set)
        self.tree.grid(row=0, column=0, sticky="nsew")
        vsb.grid(row=0, column=1, sticky="ns")
        hsb.grid(row=1, column=0, sticky="ew")
        self.rowconfigure(0, weight=1)
        self.columnconfigure(0, weight=1)
        self.tree.tag_configure("odd", background="#F6F8FB")
        self.tree.tag_configure("bad", background="#FDE9E7")
        self.tree.bind("<Double-Button-1>", self._begin_edit)
        self.editor: Optional[ttk.Entry] = None
        self.set_rows(rows)

    def set_rows(self, rows: List[List[str]]) -> None:
        self.tree.delete(*self.tree.get_children())
        for i, row in enumerate(rows):
            vals = [("" if v is None else str(v)) for v in row]
            vals += [""] * (len(self.columns) - len(vals))
            tags = ["odd"] if i % 2 else []
            if not vals[1].strip():
                tags.append("bad")
            self.tree.insert("", tk.END, values=vals[:len(self.columns)], tags=tags)

    def get_rows(self) -> List[List[str]]:
        return [list(self.tree.item(i, "values")) for i in self.tree.get_children()]

    def _begin_edit(self, event) -> None:
        row_id = self.tree.identify_row(event.y)
        col_id = self.tree.identify_column(event.x)
        if not row_id or not col_id or self.editor is not None:
            return
        idx = int(col_id.lstrip("#")) - 1
        if not (0 <= idx < len(self.columns)):
            return
        x, y, w, h = self.tree.bbox(row_id, col_id)
        column_key = self.columns[idx]
        value = self.tree.set(row_id, column_key)
        editor = ttk.Entry(self.tree, font=("微软雅黑", 10))
        editor.place(x=x, y=y, width=w, height=h)
        editor.insert(0, value)
        editor.select_range(0, tk.END)
        editor.focus_set()
        self.editor = editor

        def commit(_=None):
            if self.editor is not editor:
                return
            self.tree.set(row_id, column_key, editor.get())
            # 改完姓名后刷新行底色
            if column_key == "姓名":
                tags = [t for t in self.tree.item(row_id, "tags") if t != "bad"]
                if not editor.get().strip():
                    tags.append("bad")
                self.tree.item(row_id, tags=tags)
            editor.destroy()
            self.editor = None

        editor.bind("<Return>", commit)
        editor.bind("<FocusOut>", commit)
        editor.bind("<Escape>", lambda e: (editor.destroy(), setattr(self, "editor", None)))

    def delete_selected(self) -> int:
        ids = self.tree.selection()
        for i in ids:
            self.tree.delete(i)
        return len(ids)

    def add_row(self) -> None:
        self.tree.insert("", tk.END, values=[""] * len(self.columns), tags=["bad"])
        children = self.tree.get_children()
        if children:
            self.tree.see(children[-1])

    def count(self) -> int:
        return len(self.tree.get_children())

                return

        self.destroy()

    # -- 坐标换算 ------------------------------------------------------
    def _geom(self):
        z = float(self.zoom.get())
        return self.cell_w * z, self.cell_h * z

    def _pos_at(self, x, y):
        w, h = self._geom()
        c = int(self.canvas.canvasx(x) // w) - 1
        r = int(self.canvas.canvasy(y) // h) - 1
        if 0 <= r < self.layout.rows and 0 <= c < self.layout.cols:
            return (r, c)
        return None

    # -- 绘制 ----------------------------------------------------------
    def redraw(self) -> None:
        self.canvas.delete("all")
        w, h = self._geom()
        pad = 10
        self.canvas.create_rectangle(
            pad + w * 0.15, pad, pad + w * (self.layout.cols - 0.15), pad + h * 0.72,
            fill=COLOR["podium"], outline=COLOR["podium"])
        self.canvas.create_text(pad + w * self.layout.cols / 2, pad + h * 0.36,
                                text="讲　台　（前方）", fill="white",
                                font=("微软雅黑", 10, "bold"))
        for r in range(self.layout.rows):
            for c in range(self.layout.cols):
                value = self.layout.get(r, c)
                x0 = pad + (c + 1) * w
                y0 = pad + (r + 1) * h
                if value == SEAT:
                    fill, outline, text, fg = COLOR["normal"], COLOR["normal_border"], "座位", COLOR["text"]
                elif value == BLOCK:
                    fill, outline, text, fg = COLOR["block"], COLOR["block_border"], "过道", "#5A6B7B"
                else:
                    fill, outline, text, fg = COLOR["podium"], COLOR["podium"], "讲台", "white"
                self.canvas.create_rectangle(x0 + 3, y0 + 3, x0 + w - 3, y0 + h - 3,
                                            fill=fill, outline=outline, width=1,
                                            tags=(f"cell{r}_{c}", "cell"))
                self.canvas.create_text(x0 + w / 2, y0 + h / 2, text=text, fill=fg,
                                        font=("微软雅黑", 9), tags=(f"txt{r}_{c}", "cell"))
        # 行号
        for r in range(self.layout.rows):
            self.canvas.create_text(pad + w * 0.5, pad + (r + 1.5) * h, text=f"{r + 1}排",
                                    fill=COLOR["subtext"], font=("微软雅黑", 9))
        self.canvas.configure(scrollregion=(
            0, 0, pad * 2 + (self.layout.cols + 1.5) * w,
            pad * 2 + (self.layout.rows + 1.5) * h))
        self.info.configure(text="实时预览：" + self.layout.summary())

    def _apply(self, pos, value: str) -> None:
        if pos is None:
            return
        self.layout.set_cell(pos[0], pos[1], value)
        self.redraw()

    # -- 鼠标事件 ------------------------------------------------------
    def _press(self, event, button: int) -> None:
        self.drag_start = (event.x, event.y, self.canvas.canvasx(event.x),
                           self.canvas.canvasy(event.y))
        self.drag_button = button
        self.drag_moved = False

    def _motion(self, event) -> None:
        if not self.drag_start:
            return
        x0, y0 = self.drag_start[0], self.drag_start[1]
        if abs(event.x - x0) + abs(event.y - y0) > 8:
            self.drag_moved = True
        self.canvas.delete("rubber")
        if self.drag_moved:
            self.canvas.create_rectangle(x0, y0, event.x, event.y,
                                         outline=COLOR["select"], width=2,
                                         dash=(4, 3), tags="rubber")

    def _release(self, event, button: int) -> None:
        if not self.drag_start:
            return
        value = BLOCK if button == 3 else self.brush.get()
        if self.drag_moved:
            # 框选：把矩形覆盖到的所有格子批量改成目标类型
            w, h = self._geom()
            pad = 10
            cx0, cy0 = self.drag_start[2], self.drag_start[3]
            cx1, cy1 = self.canvas.canvasx(event.x), self.canvas.canvasy(event.y)
            c0 = int(min(cx0, cx1) // w) - 1
            c1 = int(max(cx0, cx1) // w) - 1
            r0 = int(min(cy0, cy1) // h) - 1
            r1 = int(max(cy0, cy1) // h) - 1
            changed = 0
            for r in range(max(0, r0), min(self.layout.rows - 1, r1) + 1):
                for c in range(max(0, c0), min(self.layout.cols - 1, c1) + 1):
                    self.layout.set_cell(r, c, value)
                    changed += 1
            self.redraw()
            if changed:
                self.info.configure(text=f"已批量修改 {changed} 格为「{CELL_LABEL.get(value)}」｜" +
                                    self.layout.summary())
        else:
            self._apply(self._pos_at(event.x, event.y), value)
        self.canvas.delete("rubber")
        self.drag_start = None
        self.drag_moved = False

    def _wheel(self, event) -> None:
        if event.state & 0x0004:      # 按住 Ctrl 才缩放，否则普通滚动
            z = min(1.6, max(0.6, float(self.zoom.get()) + (0.08 if event.delta > 0 else -0.08)))
            self.zoom.set(z)
            self.redraw()

    # -- 按钮 ----------------------------------------------------------
    @safe_call
    def _regenerate(self) -> None:
        rows, cols = int(self.row_var.get()), int(self.col_var.get())
        self.layout.resize(rows, cols, preserve=False)
        self.layout.name = "自定义教室"
        self.redraw()
        messagebox.showinfo("已生成", f"已生成 {rows} 行 × {cols} 列的空白网格，"
                            "请用左键点座位、右键点过道。", parent=self)

    @safe_call
    def _all_seats(self) -> None:
        self.layout.fill_all(SEAT)
        self.redraw()

    @safe_call
    def _save_template(self) -> None:
        name = simpledialog.askstring("保存教室模板", "给这个布局起个名字：",
                                      initialvalue=self.layout.name, parent=self)
        if not name:
            return
        self.layout.name = name.strip()
        folder = ClassroomLayout.template_dir(BASE_DIR)
        path = os.path.join(folder, ClassroomLayout.safe_filename(name) + ".json")
        self.layout.save(path)
        messagebox.showinfo("已保存", f"教室模板已保存：\n{path}\n下次点「加载历史模板」即可复用。",
                            parent=self)

    @safe_call
    def _load_template(self) -> None:
        paths = ClassroomLayout.list_templates(BASE_DIR)
        if not paths:
            path = filedialog.askopenfilename(
                parent=self, title="选择教室模板 JSON",
                filetypes=[("教室模板", "*.json"), ("所有文件", "*.*")])
            if not path:
                return
        else:
            picker = tk.Toplevel(self)
            picker.title("历史教室模板")
            picker.transient(self)
            picker.geometry("460x300")
            ttk.Label(picker, text="双击模板名即可加载（也可以选择其它 JSON 文件）：",
                      style="Title.TLabel").pack(padx=12, pady=(12, 6), anchor="w")
            lb = tk.Listbox(picker, font=("微软雅黑", 10))
            for p in paths:
                lb.insert(tk.END, os.path.basename(p))
            lb.pack(fill=tk.BOTH, expand=True, padx=12, pady=4)
            chosen = {"path": None}

            def pick(_=None):
                sel = lb.curselection()
                if sel:
                    chosen["path"] = paths[sel[0]]
                picker.destroy()

            def browse():
                p = filedialog.askopenfilename(parent=picker, title="选择其它模板",
                                               filetypes=[("教室模板", "*.json")])
                if p:
                    chosen["path"] = p
                picker.destroy()

            lb.bind("<Double-Button-1>", pick)
            bar = ttk.Frame(picker)
            bar.pack(fill=tk.X, padx=12, pady=(2, 12))
            ttk.Button(bar, text="浏览其它文件…", command=browse).pack(side=tk.LEFT)
            ttk.Button(bar, text="取消", command=picker.destroy).pack(side=tk.RIGHT)
            picker.grab_set()
            self.wait_window(picker)
            path = chosen["path"]
            if not path:
                return
        self.layout = ClassroomLayout.load(path)
        self.row_var.set(str(self.layout.rows))
        self.col_var.set(str(self.layout.cols))
        self.redraw()
        messagebox.showinfo("加载成功", f"已加载：{self.layout.name}\n{self.layout.summary()}", parent=self)

    @safe_call
    def _confirm(self) -> None:
        if not self.layout.seats():
            messagebox.showwarning("没有座位", "至少要留一个学生座位才能排座，请用左键点几个格子。",
                                   parent=self)
            return
        self.on_confirm(self.layout)
        self.destroy()

"""
自检脚本（selfcheck.py，可选运行，不需要图形界面）

自动造一份「很脏」的 Excel：表头不在第一行、合并单元格、干扰行、

然后跑完 导入 → 预览表往返 → 布局 → 5 套方案 → 评估 → 三种导出，

运行： python src/selfcheck.py

def make_dirty_excel(path: str) -> None:
    """构造一份故意不规整的成绩表，用来检验容错能力。"""
    from openpyxl import Workbook
    wb = Workbook()
    ws = wb.active
    ws.title = "期中成绩"
    ws.merge_cells("A1:I1")                       # 合并单元格大标题
    ws["A1"] = "六年级三班期中成绩表（表头在第 4 行）"
    ws["A3"] = "制表：教务处"
    ws["C3"] = "2026-11-07"
    headers = ["学号", "姓名", "总分", "性别", "特殊标记", "禁止同桌名单",
               "语文", "数学", "英语"]
    for c, h in enumerate(headers, start=1):
        ws.cell(row=4, column=c, value=h)

    r = 5
    for i in range(1, 37):
        score = 180 + (i * 7) % 110
        row = [f"2026{i:03d}", f"学生{i:02d}", score,
               "男" if i % 2 else "女", "", "",
               score // 3, score // 3, score - 2 * (score // 3)]
        if i == 3:
            row[4] = "重点关注"          # 浅红 + 优先靠前
        if i == 8:
            row[4] = "视力需照顾"
        if i == 5:
            row[5] = "学生06、学生07"     # 禁止同桌
        if i == 6:
            row[5] = "学生05"
        if i == 11:
            row[2] = "缺考"              # 总分缺考，但科目有分 -> 自动求和
            row[6] = 82
        if i == 12:
            row[2] = "—"                 # 缺失值
            row[7] = "N/A"
        if i == 13:
            row[2] = ""                  # 总分留空 -> 自动求和
        if i == 14:
            row[6] = "缺考"              # 单科缺考，不按 0 分
        if i == 20:
            row[2] = "缺考"              # 总分 + 三科全部缺考 -> 无总分，仍不按 0 分
            row[6] = "缺考"
            row[7] = "—"
            row[8] = "N/A"
        if i == 15:
            row[7] = -5                  # 异常分数
        if i == 16:
            row[8] = "随便填的"           # 非数值
        if i == 17:
            row[1] = ""                  # 空姓名 -> 预览表标红
        if i == 18:
            row[0] = "2026001"           # 重复学号
        if i == 19:
            row[1] = "学生01"            # 重复姓名
        for c, v in enumerate(row, start=1):
            ws.cell(row=r, column=c, value=v)
        r += 1

    r += 1                                  # 空行
    ws.cell(row=r + 1, column=1, value="合计")
    ws.cell(row=r + 1, column=3, value=9999)   # 底部合计行
    ws.cell(row=r + 1, column=7, value=9999)
    os.makedirs(os.path.dirname(path), exist_ok=True)
    wb.save(path)


def main() -> int:
    os.makedirs(OUT_DIR, exist_ok=True)
    dirty = os.path.join(OUT_DIR, "脏数据测试表.xlsx")
    make_dirty_excel(dirty)
    print(f"[1] 已构造脏 Excel：{dirty}")

    result = import_excel(dirty)
    kinds = {}
    for w in result.warnings:
        kinds[w.kind] = kinds.get(w.kind, 0) + 1
    print(f"[2] 导入成功：表头第 {result.header_row} 行，学生 {len(result.students)} 人，"
          f"忽略 {result.ignored_rows} 行，告警 {len(result.warnings)} 条 {kinds}")
    assert len(result.students) >= 34, "学生行数明显不对"
    assert result.ignored_rows >= 1, "底部合计行没有被忽略"
    assert all(s.name != "合计" and s.sid != "合计" for s in result.students), \
        "底部合计行混进了学生名单"
    assert any(s.score is None for s in result.students), "缺考没有被当成缺失"
    assert kinds.get("bad_score", 0) >= 2, "异常分数没有被告警"
    assert kinds.get("duplicate", 0) >= 1, "重复学生没有被告警"
    assert kinds.get("empty_name", 0) >= 1, "空姓名没有被告警"

    # 预览表往返（模拟老师在预览窗口改数据）
    rows = students_to_table(result.students)
    back, _ = students_from_table(rows)
    print(f"[3] 预览表双向转换：{len(rows)} 行 -> {len(back)} 名学生，正常")
    assert len(back) == len(rows)

    # 空姓名行由预览窗口负责提示，排座时会安全忽略
    clean = [s for s in back if s.name.strip()]
    # 默认教室布局（老师指定）：7 行 × 11 列，第 4 / 8 列是纵向过道
    default_layout = ClassroomLayout.default()
    assert default_layout.rows == 7 and default_layout.cols == 11, "默认布局行列数不对"
    aisles = [c for c in range(default_layout.cols)
              if default_layout.get(0, c) == BLOCK]
    assert aisles == [3, 7], f"默认布局过道列不对：{aisles}"
    print(f"[4] 默认教室布局：{default_layout.summary()}（过道在第 4、8 列）")
    layout = ClassroomLayout.default(6, 9)   # 自检继续用 6×9 覆盖小教室场景

    plans, notices = generate_plans(clean, layout, SeatingOptions(plan_count=5, spread_weight=0.7))
    attach_stats(plans, layout, clean)
    print(f"[5] 生成 {len(plans)} 套方案，全局提示 {len(notices)} 条：")
    signatures = set()
    for p in plans:
        st = p.stats
        za = st["zone_avgs"]
        sig = tuple(sorted((s.name, pos) for pos, s in p.assignment.items()))
        signatures.add(sig)
        print(f"    {p.name}: 入座 {len(p.assignment)} 人，"
              f"前/中/后排均分 {za['前排']:.1f}/{za['中排']:.1f}/{za['后排']:.1f}，"
              f"黑名单相邻 {st['ban_hits']}，结论：{st['verdict']}")
    assert len(plans) == 5
    assert len(signatures) == 5, "方案之间出现了完全相同的座位表"
    assert len(plans[0].assignment) == min(len(clean), len(layout.seats()))

    # 拖拽换座 + 撤销（用算法层接口模拟界面动作）
    a = list(plans[0].assignment.keys())[0]
    b = list(plans[0].assignment.keys())[1]
    before = (plans[0].assignment.get(a), plans[0].assignment.get(b))
    apply_swap(plans[0], a, b)
    apply_swap(plans[0], a, b)
    assert before == (plans[0].assignment.get(a), plans[0].assignment.get(b)), "换座撤销不一致"
    print("[6] 拖拽换座 / 撤销等价性校验：通过")

    # 三种导出
    xlsx = export_seating_excel(os.path.join(OUT_DIR, "示例_Excel座位表.xlsx"),
                                plans[0], layout, clean)
    png = export_canvas_png(os.path.join(OUT_DIR, "示例_座位图.png"),
                            plans[0], layout)
    report = export_report_xlsx(os.path.join(OUT_DIR, "示例_方案统计报表.xlsx"),
                                plans, layout, clean, notices)
    template = download_template(os.path.join(OUT_DIR, "示例_标准模板.xlsx"))
    for p in (xlsx, png, report, template):
        assert os.path.isfile(p) and os.path.getsize(p) > 1000, f"导出文件异常：{p}"
        print(f"[7] 导出成功：{os.path.basename(p)}（{os.path.getsize(p):,} 字节）")

    # 回归检查 ①：空位不能夹在学生中间。
    # 已入座座位必须恰好是「从前往后、从左到右」的前 N 个座位。
    seat_order = layout.seats()
    for p in plans:
        occupied = set(p.assignment)
        assert occupied == set(seat_order[:len(occupied)]), \
            f"{p.name} 出现夹在学生中间的空位"
    print("[8] 空位检查：所有方案的空位都集中在最后几排末尾，没有夹在学生中间")

    # 回归检查 ②：PNG 只含座位信息，且不给家长看老师内部标记。
    # 过道列整列删除，所以文字数 = 1 讲台 + 排号 + 保留列里的座位数。
    import matplotlib.pyplot as plt
    from matplotlib.colors import to_rgba
    draw_cols = png_columns(layout)
    cells = sum(1 for r in range(layout.rows) for c in draw_cols
                if layout.get(r, c) == SEAT)
    expected_texts = 1 + layout.rows + cells
    fig, ax = build_seat_figure(plans[0], layout)
    assert fig._suptitle is None, "PNG 里又出现了方案标题"
    assert len(ax.texts) == expected_texts, (

    assert len(ax.patches) == 1 + cells, "PNG 多画了非座位方格（过道方框？）"
    forbidden = ("过道", "图例", "浅红", "导出时间", "座位示意图", "前排",
                 "重点关注", "普通学生", "行×", "重点", "视力", "学困", "讲台格")
    extra = [t.get_text() for t in ax.texts
             if any(k in t.get_text() for k in forbidden)]
    assert not extra, f"PNG 出现附加 / 敏感文字：{extra}"
    focus_rgba = to_rgba(COLOR["focus"])
    assert not any(to_rgba(patch.get_facecolor()) == focus_rgba
                   for patch in ax.patches), "PNG 仍用浅红底泄露重点关注学生"
    plt.close(fig)
    print(f"[9] PNG 内容检查：{cells} 个座位方格（过道列已整列删除）、"
          "无过道方框、无标题图例、无重点关注标记（颜色与文字都没有）")

    print("\n✅ 全部自检通过：导入容错、预览编辑、5 套差异化方案、评估、三种导出均正常。")
    print(f"   示例文件在：{OUT_DIR}")
    return 0
